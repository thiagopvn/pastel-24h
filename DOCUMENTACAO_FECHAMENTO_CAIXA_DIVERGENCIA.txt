# DOCUMENTAÇÃO DETALHADA - FECHAMENTO DE CAIXA E DIVERGÊNCIA

## VISÃO GERAL DO SISTEMA DE DIVERGÊNCIA

O sistema de divergência de caixa é um dos componentes mais críticos da plataforma Pastel24h, responsável por identificar discrepâncias entre o dinheiro físico esperado e o dinheiro físico contado no fechamento do turno. Esta documentação explica minuciosamente como os valores de fechamento se relacionam com o campo de observação de divergência e como funciona a troca de turnos.

## CONCEITOS FUNDAMENTAIS

```typescript
// client/src/lib/calculations.ts
export interface CashFlowComponents {
  initialCash: number;      // Caixa inicial informado na abertura
  finalCash: number;        // Caixa final contado no fechamento
  cashSales: number;        // Total de vendas em dinheiro
  cashWithdrawals: number;  // Total de sangrias realizadas
  expectedCash: number;     // Caixa esperado calculado
  divergence: number;       // Diferença entre esperado e contado
}

export const calculateCashDivergence = (components: Omit<CashFlowComponents, 'expectedCash' | 'divergence'>): CashFlowComponents => {
  const expectedCash = components.initialCash + components.cashSales - components.cashWithdrawals;
  const divergence = components.finalCash - expectedCash;

  return {
    ...components,
    expectedCash,
    divergence
  };
};

// Exemplo prático:
const example = calculateCashDivergence({
  initialCash: 200.00,      // Funcionário abriu com R$ 200
  finalCash: 480.00,        // Contou R$ 480 no fechamento
  cashSales: 250.00,        // Vendeu R$ 250 em dinheiro
  cashWithdrawals: 30.00    // Fez sangria de R$ 30
});

// Resultado:
// expectedCash = 200 + 250 - 30 = 420
// divergence = 480 - 420 = +60 (sobra de R$ 60)
```

```typescript
// server/routes.ts - Endpoint de abertura de turno
app.post('/api/shifts', requireAuth, async (req, res) => {
  const { initialCash, initialCoins } = req.body;

  try {
    await db.transaction(async (tx) => {
      // 1. Buscar último turno fechado para herança
      const lastClosedShift = await tx.query.shifts.findFirst({
        where: eq(shifts.status, 'closed'),
        orderBy: [desc(shifts.id)],
        with: {
          records: { with: { product: true } },
          payments: true
        }
      });

      // 2. Calcular valores esperados
      let expectedCash = "200.00";  // Padrão do sistema
      let expectedCoins = "50.00";

      if (lastClosedShift) {
        // Usar valores definidos para próximo turno
        expectedCash = lastClosedShift.cashForNextShift || lastClosedShift.finalCash || "200.00";
        expectedCoins = lastClosedShift.coinsForNextShift || lastClosedShift.finalCoins || "50.00";

        console.log(`📋 Herdando valores do turno ${lastClosedShift.id}:`);
        console.log(`   💰 Caixa esperado: R$ ${expectedCash}`);
        console.log(`   🪙 Moedas esperadas: R$ ${expectedCoins}`);
      }

      // 3. Calcular divergência na abertura
      const informedCash = parseFloat(initialCash) || 0;
      const informedCoins = parseFloat(initialCoins) || 0;
      const expectedCashValue = parseFloat(expectedCash);
      const expectedCoinsValue = parseFloat(expectedCoins);

      const expectedTotal = expectedCashValue + expectedCoinsValue;
      const informedTotal = informedCash + informedCoins;
      const openingDiscrepancy = informedTotal - expectedTotal;

      console.log(`🧮 Calculando divergência de abertura:`);
      console.log(`   📊 Esperado: R$ ${expectedTotal.toFixed(2)}`);
      console.log(`   📋 Informado: R$ ${informedTotal.toFixed(2)}`);
      console.log(`   ⚖️  Divergência: R$ ${openingDiscrepancy.toFixed(2)}`);

      // 4. Criar novo turno
      const [newShift] = await tx.insert(shifts).values({
        userId: req.user.id,
        startTime: new Date().toISOString(),
        status: 'open',

        // Valores informados
        initialCash: informedCash.toFixed(2),
        initialCoins: informedCoins.toFixed(2),

        // Valores esperados (para referência)
        expectedCash: expectedCash,
        expectedCoins: expectedCoins,

        // Divergência na abertura
        openingDiscrepancy: openingDiscrepancy.toFixed(2)
      }).returning();

      // 5. Criar registro de pagamentos zerado
      await tx.insert(shiftPayments).values({
        shiftId: newShift.id,
        cash: '0',
        pix: '0',
        stoneCard: '0',
        stoneVoucher: '0',
        pagBankCard: '0'
      });

      // 6. Herdar produtos (sobras do turno anterior)
      if (lastClosedShift?.records) {
        const inheritanceRecords = lastClosedShift.records
          .filter(record => parseFloat(record.leftoverQty || '0') > 0)
          .map(record => ({
            shiftId: newShift.id,
            productId: record.productId,
            entryQty: record.leftoverQty,    // Sobra vira entrada
            arrivalQty: "0",
            discardQty: "0",
            consumptionQty: "0",
            leftoverQty: record.leftoverQty  // Inicialmente igual à entrada
          }));

        if (inheritanceRecords.length > 0) {
          await tx.insert(shiftRecords).values(inheritanceRecords);
          console.log(`📦 Herdados ${inheritanceRecords.length} produtos do turno anterior`);
        }
      }

      // 7. Timeline de auditoria
      await tx.insert(timeline).values({
        shiftId: newShift.id,
        userId: req.user.id,
        action: 'SHIFT_OPENED',
        description: `Turno aberto ${openingDiscrepancy !== 0 ? `com divergência de R$ ${openingDiscrepancy.toFixed(2)}` : 'sem divergência'}`,
        metadata: JSON.stringify({
          expectedCash: expectedCashValue,
          expectedCoins: expectedCoinsValue,
          informedCash,
          informedCoins,
          openingDiscrepancy,
          inheritedFrom: lastClosedShift?.id || null,
          inheritedProducts: lastClosedShift?.records?.length || 0
        })
      });

      // 8. Notificar administradores se houver divergência significativa
      if (Math.abs(openingDiscrepancy) > 10) {
        notifyAdminClients('shift-opened-with-divergence', {
          shiftId: newShift.id,
          userId: req.user.id,
          userName: req.user.name,
          openingDiscrepancy,
          timestamp: new Date().toISOString()
        });
      }

      res.status(201).json({
        ...newShift,
        openingDiscrepancy
      });
    });

  } catch (error) {
    console.error('❌ Erro ao abrir turno:', error);
    res.status(500).json({ error: 'Erro interno do servidor' });
  }
});
```

```typescript
// server/routes.ts - Endpoint de fechamento
app.post('/api/shifts/:id/close', requireAuth, async (req, res) => {
  const shiftId = parseInt(req.params.id);
  const {
    finalCash,
    finalCoins,
    envelopeCash,
    envelopeCoins,
    payments,
    cumulativeCardValues,
    observations
  } = req.body;

  try {
    await db.transaction(async (tx) => {
      // 1. Buscar dados completos do turno
      const shift = await tx.query.shifts.findFirst({
        where: and(eq(shifts.id, shiftId), eq(shifts.status, 'open')),
        with: {
          user: true,
          payments: true,
          records: { with: { product: true } }
        }
      });

      if (!shift) {
        throw new Error('Turno não encontrado ou já fechado');
      }

      // 2. Processar pagamentos (incluindo cálculo de cartões se necessário)
      let finalPayments = payments;

      if (cumulativeCardValues) {
        const cardCalculator = new CardPaymentCalculator(tx);
        const calculatedCards = await cardCalculator.calculateRealCardPayments(
          shiftId,
          { ...payments, ...cumulativeCardValues }
        );

        finalPayments = {

          stoneVoucher: calculatedCards.realValues.stoneVoucher.toFixed(2),
          pagBankCard: calculatedCards.realValues.pagBankCard.toFixed(2)
        };

        await tx.update(shiftPayments)
          .set({
            ...finalPayments,
            stoneCardCumulative: cumulativeCardValues.stoneCardCumulative,
            stoneVoucherCumulative: cumulativeCardValues.stoneVoucherCumulative,
            pagBankCardCumulative: cumulativeCardValues.pagBankCardCumulative,
            calculatedFromCumulative: true
          })
          .where(eq(shiftPayments.shiftId, shiftId));
      } else {
        await tx.update(shiftPayments)
          .set(finalPayments)
          .where(eq(shiftPayments.shiftId, shiftId));
      }

      // 3. Calcular todos os valores financeiros
      const actualCash = parseFloat(finalCash) || 0;
      const actualCoins = parseFloat(finalCoins) || 0;
      const envelopeCashAmount = parseFloat(envelopeCash) || 0;
      const envelopeCoinsAmount = parseFloat(envelopeCoins) || 0;

      // Valores para próximo turno
      const cashForNext = actualCash - envelopeCashAmount;
      const coinsForNext = actualCoins - envelopeCoinsAmount;

      // 4. CÁLCULO DETALHADO DA DIVERGÊNCIA
      const initialCash = parseFloat(shift.initialCash || '0');
      const cashSales = parseFloat(finalPayments.cash || '0');

      // Buscar total de sangrias
      const withdrawalsResult = await tx.select({ total: sum(cashAdjustments.amount) })
        .from(cashAdjustments)
        .where(and(
          eq(cashAdjustments.shiftId, shiftId),
          eq(cashAdjustments.type, 'withdrawal')
        ));

      const totalWithdrawals = withdrawalsResult[0]?.total || 0;

      // Cálculo da divergência
      const expectedCash = initialCash + cashSales - totalWithdrawals;
      const cashDivergence = actualCash - expectedCash;

      console.log('🧮 Calculando divergência de fechamento:');
      console.log(`   🏦 Caixa inicial: R$ ${initialCash.toFixed(2)}`);
      console.log(`   💰 Vendas em dinheiro: R$ ${cashSales.toFixed(2)}`);
      console.log(`   📤 Sangrias: R$ ${totalWithdrawals.toFixed(2)}`);
      console.log(`   🎯 Caixa esperado: R$ ${expectedCash.toFixed(2)}`);
      console.log(`   📊 Caixa contado: R$ ${actualCash.toFixed(2)}`);
      console.log(`   ⚖️  Divergência: R$ ${cashDivergence.toFixed(2)}`);

      // 5. Fechar turno com todos os dados
      await tx.update(shifts).set({
        status: 'closed',
        endTime: new Date().toISOString(),

        // Valores finais contados
        finalCash: actualCash.toFixed(2),
        finalCoins: actualCoins.toFixed(2),

        // Envelope
        envelopeCash: envelopeCashAmount.toFixed(2),
        envelopeCoins: envelopeCoinsAmount.toFixed(2),

        coinsForNextShift: coinsForNext.toFixed(2),

        cashDivergence: cashDivergence.toFixed(2),

        // Observações
        closingObservations: observations || null

      // 6. Finalizar colaboradores
      await tx.update(shiftCollaborators)
        .set({
          endTime: new Date().toISOString(),
          hoursWorked: sql`(julianday(${new Date().toISOString()}) - julianday(start_time)) * 24`
        })
        .where(and(
          eq(shiftCollaborators.shiftId, shiftId),
          isNull(shiftCollaborators.endTime)
        ));

      // 7. Timeline detalhada de auditoria
      await tx.insert(timeline).values({

        userId: req.user.id,
        action: 'SHIFT_CLOSED',
        description: `Turno fechado ${Math.abs(cashDivergence) > 0.01

          closedAt: new Date().toISOString()
        })

        notifyAdminClients('shift-closed-with-divergence', {

        console.log(`   💰 Divergência: R$ ${cashDivergence.toFixed(2)}`);
        console.log(`   📝 Observações: ${observations || 'Nenhuma'}`);
      } else {
        // Fechamento balanceado
        notifyAdminClients('shift-closed-balanced', {
          shiftId,
          userId: shift.userId,
          userName: shift.user.name,
          totalPayments: Object.values(finalPayments).reduce((sum, val) => sum + parseFloat(val), 0),
          closedAt: new Date().toISOString()
        });
      }

      res.json({
        message: 'Turno fechado com sucesso',
        shiftId,
        cashDivergence,
        expectedCash,
        actualCash,
        cashForNext,
        coinsForNext
      });
    });

  } catch (error) {
    console.error('❌ Erro ao fechar turno:', error);
    res.status(500).json({ error: error.message || 'Erro interno do servidor' });
  }
});
```

```typescript
export class ShiftInheritanceService {
  constructor(private db: any) {}

  async calculateInheritanceValues(lastClosedShiftId?: number): Promise<InheritanceData> {
    if (!lastClosedShiftId) {
      // Buscar último turno fechado
      const lastShift = await this.db.query.shifts.findFirst({
        where: eq(shifts.status, 'closed'),
        orderBy: [desc(shifts.id)],
        with: {
          records: { with: { product: true } }
        }
      });

      if (!lastShift) {
        return this.getDefaultInheritanceValues();
      }

      lastClosedShiftId = lastShift.id;
    }

    const lastShift = await this.db.query.shifts.findFirst({
      where: eq(shifts.id, lastClosedShiftId),
      with: {
        records: { with: { product: true } }
      }
    });

    if (!lastShift) {
      return this.getDefaultInheritanceValues();
    }

    // 1. HERANÇA DE DINHEIRO
    const cashInheritance = this.calculateCashInheritance(lastShift);

    // 2. HERANÇA DE PRODUTOS
    const productInheritance = this.calculateProductInheritance(lastShift.records);

    return {
      cash: cashInheritance,
      products: productInheritance,
      source: {
        shiftId: lastShift.id,
        closedAt: lastShift.endTime,
        closedBy: lastShift.userId
      }
    };
  }

  private calculateCashInheritance(lastShift: any): CashInheritanceData {
    // Usar valores específicos para próximo turno se definidos
    const cashForNext = parseFloat(lastShift.cashForNextShift || '0');
    const coinsForNext = parseFloat(lastShift.coinsForNextShift || '0');

    if (cashForNext > 0 || coinsForNext > 0) {
      return {
        expectedCash: cashForNext.toFixed(2),
        expectedCoins: coinsForNext.toFixed(2),
        source: 'calculated', // Calculado no fechamento anterior
        calculation: {
          finalCash: parseFloat(lastShift.finalCash || '0'),
          finalCoins: parseFloat(lastShift.finalCoins || '0'),
          envelopeCash: parseFloat(lastShift.envelopeCash || '0'),
          envelopeCoins: parseFloat(lastShift.envelopeCoins || '0'),
          remaining: {
            cash: cashForNext,
            coins: coinsForNext
          }
        }
      };
    }

    // Fallback: usar valores finais do turno anterior
    return {
      expectedCash: lastShift.finalCash || '200.00',
      expectedCoins: lastShift.finalCoins || '50.00',
      source: 'fallback', // Usando valores finais como fallback
      calculation: {
        finalCash: parseFloat(lastShift.finalCash || '200'),
        finalCoins: parseFloat(lastShift.finalCoins || '50'),
        envelopeCash: 0,
        envelopeCoins: 0,
        remaining: {
          cash: parseFloat(lastShift.finalCash || '200'),
          coins: parseFloat(lastShift.finalCoins || '50')
        }
      }
    };
  }

  private calculateProductInheritance(records: any[]): ProductInheritanceData[] {
    return records
      .filter(record => parseFloat(record.leftoverQty || '0') > 0)
      .map(record => ({
        productId: record.productId,
        productName: record.product.name,
        category: record.product.category,
        leftoverQty: parseFloat(record.leftoverQty || '0'),
        entryQtyForNext: parseFloat(record.leftoverQty || '0'), // Sobra vira entrada
        lastShiftData: {
          entryQty: parseFloat(record.entryQty || '0'),
          arrivalQty: parseFloat(record.arrivalQty || '0'),
          soldQty: this.calculateSoldQty(record),
          discardQty: parseFloat(record.discardQty || '0'),
          consumptionQty: parseFloat(record.consumptionQty || '0')
        }
      }));
  }

  private calculateSoldQty(record: any): number {
    const entry = parseFloat(record.entryQty || '0');
    const arrival = parseFloat(record.arrivalQty || '0');
    const leftover = parseFloat(record.leftoverQty || '0');
    const discard = parseFloat(record.discardQty || '0');
    const consumption = parseFloat(record.consumptionQty || '0');

    return entry + arrival - leftover - discard - consumption;
  }

  private getDefaultInheritanceValues(): InheritanceData {
    return {
      cash: {
        expectedCash: '200.00',
        expectedCoins: '50.00',
        source: 'default',
        calculation: {
          finalCash: 200,
          finalCoins: 50,
          envelopeCash: 0,
          envelopeCoins: 0,
          remaining: { cash: 200, coins: 50 }
        }
      },
      products: [],
      source: null
    };
  }
}
```

```typescript
// server/lib/shift-validation.ts
export class ShiftValidationService {
  constructor(private db: any) {}

  async validateShiftTransition(
    lastShiftId: number,
    newShiftData: { initialCash: number; initialCoins: number }
  ): Promise<ValidationResult> {
    const validationResult: ValidationResult = {
      isValid: true,
      warnings: [],
      errors: [],
      divergences: []
    };

    // 1. Buscar dados do último turno
    const lastShift = await this.db.query.shifts.findFirst({
      where: eq(shifts.id, lastShiftId),
      with: { payments: true }
    });

    if (!lastShift) {
      validationResult.warnings.push('Nenhum turno anterior encontrado para validação');
      return validationResult;
    }

    // 2. Validar herança de dinheiro
    const expectedCash = parseFloat(lastShift.cashForNextShift || lastShift.finalCash || '200');
    const expectedCoins = parseFloat(lastShift.coinsForNextShift || lastShift.finalCoins || '50');

    const cashDiscrepancy = newShiftData.initialCash - expectedCash;
    const coinsDiscrepancy = newShiftData.initialCoins - expectedCoins;

    if (Math.abs(cashDiscrepancy) > 0.01) {
      validationResult.divergences.push({
        type: 'cash',
        expected: expectedCash,
        actual: newShiftData.initialCash,
        difference: cashDiscrepancy,
        severity: Math.abs(cashDiscrepancy) > 10 ? 'high' : 'medium'
      });
    }

    if (Math.abs(coinsDiscrepancy) > 0.01) {
      validationResult.divergences.push({
        type: 'coins',
        expected: expectedCoins,
        actual: newShiftData.initialCoins,
        difference: coinsDiscrepancy,
        severity: Math.abs(coinsDiscrepancy) > 5 ? 'high' : 'medium'
      });
    }

    // 3. Validar timing entre turnos
    const lastShiftEnd = new Date(lastShift.endTime);
    const now = new Date();
    const timeDifference = now.getTime() - lastShiftEnd.getTime();
    const hoursDifference = timeDifference / (1000 * 60 * 60);

    if (hoursDifference > 24) {
      validationResult.warnings.push(
        `Último turno foi fechado há ${hoursDifference.toFixed(1)} horas. Verifique se os valores ainda são válidos.`
      );
    }

    // 4. Validar se último turno teve divergências
    const lastDivergence = parseFloat(lastShift.cashDivergence || '0');
    if (Math.abs(lastDivergence) > 0.01) {
      validationResult.warnings.push(
        `Turno anterior teve divergência de R$ ${lastDivergence.toFixed(2)}. Verificar se foi resolvida.`
      );
    }

    // 5. Determinar se validação geral é válida
    const hasHighSeverityIssues = validationResult.divergences.some(d => d.severity === 'high');
    if (hasHighSeverityIssues) {
      validationResult.isValid = false;
      validationResult.errors.push('Divergências de alta severidade detectadas. Revisão necessária.');
    }

    return validationResult;
  }
}
```

```typescript
// server/lib/divergence-report.ts
export class DivergenceReportService {
  constructor(private db: any) {}

  async generateDivergenceReport(
    startDate: string,
    endDate: string,
    options: ReportOptions = {}
  ): Promise<DivergenceReport> {

    // 1. Buscar turnos com divergência no período
    const shiftsWithDivergence = await this.db.query.shifts.findMany({
      where: and(
        gte(shifts.startTime, startDate),
        lte(shifts.startTime, endDate),
        ne(shifts.cashDivergence, '0.00')
      ),
      with: {
        user: true,
        payments: true
      },
      orderBy: [desc(shifts.startTime)]
    });

    // 2. Analisar padrões de divergência
    const analysis = this.analyzeDivergencePatterns(shiftsWithDivergence);

    // 3. Calcular estatísticas
    const statistics = this.calculateDivergenceStatistics(shiftsWithDivergence);

    // 4. Identificar funcionários com divergências recorrentes
    const employeeAnalysis = this.analyzeEmployeeDivergences(shiftsWithDivergence);

    return {
      period: { start: startDate, end: endDate },
      totalShifts: shiftsWithDivergence.length,
      statistics,
      patterns: analysis,
      employeeAnalysis,
      shifts: shiftsWithDivergence.map(shift => this.formatShiftForReport(shift)),
      recommendations: this.generateRecommendations(analysis, employeeAnalysis)
    };
  }

  private analyzeDivergencePatterns(shifts: any[]): DivergencePatterns {
    const patterns: DivergencePatterns = {
      byTimeOfDay: {},
      byDayOfWeek: {},
      byAmount: { small: 0, medium: 0, large: 0 },
      trends: []
    };

    shifts.forEach(shift => {
      const divergence = parseFloat(shift.cashDivergence || '0');
      const shiftDate = new Date(shift.startTime);

      // Padrão por hora do dia
      const hour = shiftDate.getHours();
      const timeSlot = this.getTimeSlot(hour);
      patterns.byTimeOfDay[timeSlot] = (patterns.byTimeOfDay[timeSlot] || 0) + 1;

      // Padrão por dia da semana
      const dayOfWeek = shiftDate.getDay();
      const dayName = ['Domingo', 'Segunda', 'Terça', 'Quarta', 'Quinta', 'Sexta', 'Sábado'][dayOfWeek];
      patterns.byDayOfWeek[dayName] = (patterns.byDayOfWeek[dayName] || 0) + 1;

      // Padrão por magnitude da divergência
      const absDiv = Math.abs(divergence);
      if (absDiv <= 5) {
        patterns.byAmount.small++;
      } else if (absDiv <= 20) {
        patterns.byAmount.medium++;
      } else {
        patterns.byAmount.large++;
      }
    });

    return patterns;
  }

  private calculateDivergenceStatistics(shifts: any[]): DivergenceStatistics {
    const divergences = shifts.map(s => parseFloat(s.cashDivergence || '0'));

    const total = divergences.reduce((sum, div) => sum + Math.abs(div), 0);
    const average = divergences.length > 0 ? total / divergences.length : 0;

    const positive = divergences.filter(d => d > 0);
    const negative = divergences.filter(d => d < 0);

    return {
      totalDivergence: total,
      averageDivergence: average,
      maxPositive: positive.length > 0 ? Math.max(...positive) : 0,
      maxNegative: negative.length > 0 ? Math.min(...negative) : 0,
      positiveCount: positive.length,
      negativeCount: negative.length,
      distribution: {
        under5: divergences.filter(d => Math.abs(d) <= 5).length,
        from5to20: divergences.filter(d => Math.abs(d) > 5 && Math.abs(d) <= 20).length,
        over20: divergences.filter(d => Math.abs(d) > 20).length
      }
    };
  }

  private formatShiftForReport(shift: any): FormattedShiftDivergence {
    const divergence = parseFloat(shift.cashDivergence || '0');

    return {
      id: shift.id,
      date: shift.startTime,
      employee: shift.user.name,
      expectedCash: parseFloat(shift.expectedFinalCash || '0'),
      actualCash: parseFloat(shift.finalCash || '0'),
      divergence,
      divergenceType: divergence > 0 ? 'sobra' : divergence < 0 ? 'falta' : 'balanceado',
      severity: this.getDivergenceSeverity(Math.abs(divergence)),
      observations: shift.closingObservations,
      cashSales: parseFloat(shift.payments?.cash || '0'),
      totalSales: this.calculateTotalSales(shift.payments)
    };
  }

  private getDivergenceSeverity(absDivergence: number): 'low' | 'medium' | 'high' {
    if (absDivergence <= 5) return 'low';
    if (absDivergence <= 20) return 'medium';
    return 'high';
  }

  private generateRecommendations(
    patterns: DivergencePatterns,
    employeeAnalysis: EmployeeAnalysis[]
  ): string[] {
    const recommendations: string[] = [];

    // Recomendações baseadas em padrões temporais
    const problematicTimes = Object.entries(patterns.byTimeOfDay)
      .filter(([time, count]) => count > 3)
      .map(([time]) => time);

    if (problematicTimes.length > 0) {
      recommendations.push(
        `Revisar procedimentos nos horários: ${problematicTimes.join(', ')}. Maior incidência de divergências.`
      );
    }

    // Recomendações baseadas em funcionários
    const problematicEmployees = employeeAnalysis
      .filter(emp => emp.averageDivergence > 10 || emp.divergenceRate > 0.3)
      .map(emp => emp.employeeName);

    if (problematicEmployees.length > 0) {
      recommendations.push(
        `Treinamento adicional recomendado para: ${problematicEmployees.join(', ')}.`
      );
    }

    // Recomendações baseadas em magnitude
    if (patterns.byAmount.large > patterns.byAmount.small) {
      recommendations.push(
        'Alto número de divergências grandes. Revisar processo de contagem e sangrias.'
      );
    }

    return recommendations;
  }
}
```

```typescript
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Data</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Funcionário</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Esperado</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Contado</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Divergência</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Observações</th>
                <th className="px-4 py-3 text-left text-sm font-medium text-gray-900">Ações</th>
              </tr>
            </thead>
            <tbody>
              {filteredShifts.map((shift: any) => (

                    {new Date(shift.date).toLocaleString()}

                  <td className="px-4 py-3 text-sm font-medium">

                    R$ {shift.actualCash.toFixed(2)}
                  </td>
                  <td className="px-4 py-3 text-sm">
                    <span className={`inline-flex px-2 py-1 rounded-full text-xs font-medium ${
                      shift.divergence > 0
                        ? 'bg-yellow-100 text-yellow-800'
                        : shift.divergence < 0
                        ? 'bg-red-100 text-red-800'
                        : 'bg-green-100 text-green-800'
                    }`}>
                      R$ {shift.divergence.toFixed(2)}
                      {shift.divergence > 0 && ' (sobra)'}
                      {shift.divergence < 0 && ' (falta)'}
                    </span>
                  </td>
                  <td className="px-4 py-3 text-sm max-w-xs truncate">
                    {shift.observations || '-'}
                  </td>
                  <td className="px-4 py-3 text-sm">
                    <button
                      onClick={() => handleViewShiftDetails(shift.id)}
                      className="text-blue-600 hover:text-blue-800"
                    >
                      Ver Detalhes
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};
```
