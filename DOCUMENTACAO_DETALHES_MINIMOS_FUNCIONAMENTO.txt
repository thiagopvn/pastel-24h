import passport from 'passport';

// 2. Configuração da estratégia local
passport.use(new LocalStrategy({
    usernameField: 'email',
    passwordField: 'password',
    passReqToCallback: true  // Passar req para callback
  }, async (req, email, password, done) => {
    const startTime = Date.now();

    try {
      console.log(`🔍 Tentativa de login para: ${email} (IP: ${req.ip})`);

        where: eq(users.email, email.toLowerCase().trim())

        console.log(`❌ Usuário não encontrado: ${email}`);

        await new Promise(resolve => setTimeout(resolve, 1000));

        return done(null, false, {
          message: 'Email não encontrado'
        });

      // 4. Verificar senha com bcrypt
      const isValidPassword = await bcrypt.compare(password, user.password);

            ip: req.ip,
            userAgent: req.headers['user-agent'],
            timestamp: new Date().toISOString()
          })
        });

        return done(null, false, {
          message: 'Senha incorreta'
        });
      }

      // 5. Login bem-sucedido
      const loginTime = Date.now() - startTime;
      console.log(`✅ Login bem-sucedido para: ${email} (${loginTime}ms)`);

    }
  }
));

// client/src/lib/queryClient.ts
import { QueryClient, QueryCache, MutationCache } from '@tanstack/react-query';

// 1. Configuração de cache personalizada
const queryCache = new QueryCache({
  onError: (error, query) => {
    console.error(`🚨 Erro na query ${query.queryKey}:`, error);

    // Tratar erros específicos
    if (error instanceof Error) {
      if (error.message.includes('Network Error')) {
        toast.error('Erro de conexão. Verifique sua internet.');
      } else if (error.message.includes('401')) {
        // Redirecionar para login se não autorizado
        window.location.href = '/login';
      }
    }
  },
  onSuccess: (data, query) => {
    console.log(`✅ Query bem-sucedida ${query.queryKey}`);
  }
});

const mutationCache = new MutationCache({
  onError: (error, variables, context, mutation) => {
    console.error(`🚨 Erro na mutation:`, error);
  },
  onSuccess: (data, variables, context, mutation) => {
    console.log(`✅ Mutation bem-sucedida`);
  }
});

// 2. Query client com configurações otimizadas
export const queryClient = new QueryClient({
  queryCache,
  mutationCache,
  defaultOptions: {
    queries: {
      // Cache por 5 minutos
      staleTime: 5 * 60 * 1000,

      // Manter cache por 10 minutos após ficar "stale"
      gcTime: 10 * 60 * 1000,

      // Retry com backoff exponencial
      retry: (failureCount, error) => {
        if (error instanceof Error && error.message.includes('401')) {
          return false; // Não tentar novamente se não autorizado
        }
        return failureCount < 3;
      },

      retryDelay: (attemptIndex) => Math.min(1000 * 2 ** attemptIndex, 30000),

      // Não refetch em foco por padrão
      refetchOnWindowFocus: false,

      // Refetch em reconexão
      refetchOnReconnect: true,

      // Não refetch em mount se dados estão frescos
      refetchOnMount: (query) => {
        return Date.now() - query.dataUpdatedAt > 2 * 60 * 1000; // 2 minutos
      }
    },
    mutations: {
      // Retry mutations apenas para erros de rede
      retry: (failureCount, error) => {
        if (error instanceof Error && error.message.includes('Network')) {
          return failureCount < 2;
        }
        return false;
      }
    }
  }
});

// client/src/hooks/useShiftData.ts
export const useShiftData = (shiftId: number | null) => {
  // 1. Query principal do turno
  const shiftQuery = useQuery({
    queryKey: ['shift', shiftId],
    queryFn: () => api.get(`/api/shifts/${shiftId}`),
    enabled: !!shiftId,
    staleTime: 2 * 60 * 1000, // 2 minutos - dados financeiros precisam ser frescos

    // Comparador personalizado para evitar re-renders desnecessários
    select: (data) => ({
      ...data,
      // Arredondar valores para evitar flutuação de ponto flutuante
      initialCash: Math.round(data.initialCash * 100) / 100,
      finalCash: data.finalCash ? Math.round(data.finalCash * 100) / 100 : null
    }),

    // Placeholder data enquanto carrega
    placeholderData: (previousData) => previousData
  });

  // 2. Query para registros de inventário
  const recordsQuery = useQuery({
    queryKey: ['shift-records', shiftId],
    queryFn: () => api.get(`/api/shifts/${shiftId}/records`),
    enabled: !!shiftId,

    // Transformar dados para facilitar uso no componente
    select: (data) => {
      return data.map((record: any) => ({
        ...record,
        soldQty: calculateSoldQuantity(
          parseFloat(record.entryQty || '0'),
          parseFloat(record.arrivalQty || '0'),
          parseFloat(record.leftoverQty || '0'),
          parseFloat(record.discardQty || '0'),
          parseFloat(record.consumptionQty || '0')
        )
      }));
    }
  });

  // 3. Query para pagamentos
  const paymentsQuery = useQuery({
    queryKey: ['shift-payments', shiftId],
    queryFn: () => api.get(`/api/shifts/${shiftId}/payments`),
    enabled: !!shiftId,
    staleTime: 1 * 60 * 1000, // 1 minuto - pagamentos mudam frequentemente
  });

  // 4. Computed values baseados nas queries
  const computedData = useMemo(() => {
    if (!shiftQuery.data || !recordsQuery.data || !paymentsQuery.data) {
      return null;
    }

    const totalRevenue = recordsQuery.data.reduce((sum: number, record: any) => {
      return sum + (record.soldQty * record.product.price);
    }, 0);

    const totalPayments = Object.values(paymentsQuery.data).reduce((sum: number, payment: any) => {
      return sum + parseFloat(payment || '0');
    }, 0);

    const cashDivergence = calculateCashDivergence({
      initialCash: shiftQuery.data.initialCash,
      finalCash: shiftQuery.data.finalCash || 0,
      cashSales: parseFloat(paymentsQuery.data.cash || '0'),
      cashWithdrawals: 0 // TODO: Buscar sangrias
    });

    return {
      totalRevenue,
      totalPayments,
      cashDivergence,
      isBalanced: Math.abs(cashDivergence) < 0.01
    };
  }, [shiftQuery.data, recordsQuery.data, paymentsQuery.data]);

  // 5. Status consolidado
  const isLoading = shiftQuery.isLoading || recordsQuery.isLoading || paymentsQuery.isLoading;
  const isError = shiftQuery.isError || recordsQuery.isError || paymentsQuery.isError;
  const error = shiftQuery.error || recordsQuery.error || paymentsQuery.error;

  return {
    shift: shiftQuery.data,
    records: recordsQuery.data,
    payments: paymentsQuery.data,
    computed: computedData,
    isLoading,
    isError,
    error,
    refetch: () => {
      shiftQuery.refetch();
      recordsQuery.refetch();
      paymentsQuery.refetch();
    }
  };
};

// client/src/hooks/useWebSocketWithReconnect.ts
export const useWebSocketWithReconnect = (url: string, options: WebSocketOptions = {}) => {
  const [socket, setSocket] = useState<WebSocket | null>(null);
  const [connectionStatus, setConnectionStatus] = useState<ConnectionStatus>('DISCONNECTED');
  const [messageHistory, setMessageHistory] = useState<MessageHistoryItem[]>([]);

  const reconnectTimeoutRef = useRef<NodeJS.Timeout | null>(null);
  const reconnectAttemptRef = useRef(0);
  const maxReconnectAttempts = useRef(options.maxReconnectAttempts || 5);
  const reconnectIntervalRef = useRef(options.reconnectInterval || 3000);

  const queryClient = useQueryClient();

  // 1. Função de conexão
  const connect = useCallback(() => {
    if (socket?.readyState === WebSocket.OPEN) {
      console.log('⚠️  WebSocket já conectado');
      return;
    }

    console.log(`🔌 Tentando conectar WebSocket... (tentativa ${reconnectAttemptRef.current + 1})`);
    setConnectionStatus('CONNECTING');

    const ws = new WebSocket(url);

    // 2. Event handlers
    ws.onopen = (event) => {
      console.log('✅ WebSocket conectado');
      setConnectionStatus('CONNECTED');
      setSocket(ws);
      reconnectAttemptRef.current = 0;

      // Callback de conexão bem-sucedida
      options.onConnect?.(event);

      // Enviar mensagem de identificação se fornecida
      if (options.identificationMessage) {
        ws.send(JSON.stringify(options.identificationMessage));
      }
    };

    ws.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data);

        // 3. Adicionar ao histórico
        setMessageHistory(prev => [
          {
            id: Date.now(),
            timestamp: new Date(),
            type: 'received',
            data: message
          },
          ...prev.slice(0, 99) // Manter últimas 100 mensagens
        ]);

        // 4. Processar mensagem
        handleMessage(message);

        // Callback de mensagem recebida
        options.onMessage?.(message);

      } catch (error) {
        console.error('🚨 Erro ao processar mensagem WebSocket:', error);
        options.onError?.(error);
      }
    };

    ws.onclose = (event) => {
      console.log(`🔌 WebSocket desconectado (código: ${event.code}, motivo: ${event.reason})`);
      setConnectionStatus('DISCONNECTED');
      setSocket(null);

      // 5. Lógica de reconexão
      if (event.code !== 1000 && reconnectAttemptRef.current < maxReconnectAttempts.current) {
        const delay = Math.min(
          reconnectIntervalRef.current * Math.pow(2, reconnectAttemptRef.current),
          30000 // Máximo 30 segundos
        );

        console.log(`🔄 Tentando reconectar em ${delay}ms...`);

        reconnectTimeoutRef.current = setTimeout(() => {
          reconnectAttemptRef.current++;
          connect();
        }, delay);
      } else {
        console.log('❌ Máximo de tentativas de reconexão atingido');
        setConnectionStatus('FAILED');
        options.onReconnectFailed?.();
      }

      options.onDisconnect?.(event);
    };

    ws.onerror = (error) => {
      console.error('🚨 Erro WebSocket:', error);
      setConnectionStatus('ERROR');
      options.onError?.(error);
    };

  }, [url, options, socket]);

  // 6. Função para processar mensagens específicas
  const handleMessage = (message: any) => {
    switch (message.type) {
      case 'shift-updated':
        // Invalidar queries relacionadas ao turno
        queryClient.invalidateQueries(['shift', message.shiftId]);
        queryClient.invalidateQueries(['shift-records', message.shiftId]);
        queryClient.invalidateQueries(['shift-payments', message.shiftId]);
        break;

      case 'inventory-updated':
        queryClient.invalidateQueries(['shift-records', message.shiftId]);
        break;

      case 'payment-added':
        queryClient.invalidateQueries(['shift-payments', message.shiftId]);
        break;

      case 'admin-notification':
        // Processar notificação administrativa
        if (message.priority === 'high') {
          toast.warning(message.title, {
            description: message.message,
            duration: 10000
          });
        }
        break;

      case 'heartbeat':
        // Responder ao heartbeat
        sendMessage({ type: 'heartbeat-response', timestamp: Date.now() });
        break;
    }
  };

  // 7. Função para enviar mensagens
  const sendMessage = useCallback((message: any) => {
    if (socket?.readyState === WebSocket.OPEN) {
      const messageWithTimestamp = {
        ...message,
        timestamp: new Date().toISOString(),
        id: Date.now()
      };

      socket.send(JSON.stringify(messageWithTimestamp));

      // Adicionar ao histórico
      setMessageHistory(prev => [
        {
          id: Date.now(),
          timestamp: new Date(),
          type: 'sent',
          data: messageWithTimestamp
        },
        ...prev.slice(0, 99)
      ]);

      return true;
    } else {
      console.warn('⚠️  WebSocket não conectado. Mensagem não enviada:', message);
      return false;
    }
  }, [socket]);

  // 8. Função para desconectar
  const disconnect = useCallback(() => {
    if (reconnectTimeoutRef.current) {
      clearTimeout(reconnectTimeoutRef.current);
      reconnectTimeoutRef.current = null;
    }

    if (socket) {
      socket.close(1000, 'Desconexão intencional');
    }

    setConnectionStatus('DISCONNECTED');
    setSocket(null);
  }, [socket]);

  // 9. Effect para conectar/desconectar
  useEffect(() => {
    connect();

    return () => {
      disconnect();
    };
  }, [connect, disconnect]);

  // 10. Cleanup no unmount
  useEffect(() => {
    return () => {
      if (reconnectTimeoutRef.current) {
        clearTimeout(reconnectTimeoutRef.current);
      }
    };
  }, []);

  return {
    socket,
    connectionStatus,
    messageHistory,
    sendMessage,
    connect,
    disconnect,
    isConnected: connectionStatus === 'CONNECTED',
    isConnecting: connectionStatus === 'CONNECTING',
    reconnectAttempts: reconnectAttemptRef.current
  };
};

interface ErrorBoundaryState {
  hasError: boolean;
  error?: Error;
  errorInfo?: ErrorInfo;
}

export class ErrorBoundary extends Component<
  { children: ReactNode; fallback?: ComponentType<{ error: Error }> },
  ErrorBoundaryState
> {
  constructor(props: any) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('🚨 Error Boundary capturou erro:', error, errorInfo);

    // 1. Log detalhado do erro
    const errorDetails = {
      message: error.message,
      stack: error.stack,
      componentStack: errorInfo.componentStack,
      timestamp: new Date().toISOString(),
      userAgent: navigator.userAgent,
      url: window.location.href
    };

    // 2. Enviar erro para serviço de monitoramento (se configurado)
    this.reportError(errorDetails);

    // 3. Salvar no localStorage para debug
    try {
      const savedErrors = JSON.parse(localStorage.getItem('app_errors') || '[]');
      savedErrors.push(errorDetails);
      localStorage.setItem('app_errors', JSON.stringify(savedErrors.slice(-10))); // Últimos 10
    } catch (e) {
      console.warn('Não foi possível salvar erro no localStorage');
    }

    this.setState({ error, errorInfo });
  }

  private reportError = async (errorDetails: any) => {
    try {
      // Enviar para API de erro (implementar se necessário)
      await fetch('/api/errors', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(errorDetails)
      });
    } catch (e) {
      console.warn('Não foi possível enviar erro para servidor');
    }
  };

  render() {
    if (this.state.hasError) {
      const FallbackComponent = this.props.fallback || DefaultErrorFallback;
      return <FallbackComponent error={this.state.error!} />;
    }

    return this.props.children;
  }

// shared/logger.ts
enum LogLevel {
  DEBUG = 0,
  INFO = 1,
  WARN = 2,
  ERROR = 3
}

interface LogEntry {
  level: LogLevel;
  message: string;
  timestamp: string;
  context?: any;
  userId?: number;
  sessionId?: string;
}

class Logger {
  private logLevel: LogLevel;
  private buffer: LogEntry[] = [];
  private maxBufferSize = 100;

  constructor(level: LogLevel = LogLevel.INFO) {
    this.logLevel = level;
  }

  private log(level: LogLevel, message: string, context?: any) {
    if (level < this.logLevel) return;

    const entry: LogEntry = {
      level,
      message,
      timestamp: new Date().toISOString(),
      context,
      userId: this.getCurrentUserId(),
      sessionId: this.getSessionId()
    };

    // 1. Console output com cores
    const colors = {
      [LogLevel.DEBUG]: '\x1b[36m', // Cyan
      [LogLevel.INFO]: '\x1b[32m',  // Green
      [LogLevel.WARN]: '\x1b[33m',  // Yellow
      [LogLevel.ERROR]: '\x1b[31m'  // Red
    };

    const reset = '\x1b[0m';
    const levelName = LogLevel[level];

    console.log(
      `${colors[level]}[${levelName}]${reset} ${entry.timestamp} - ${message}`,
      context ? context : ''
    );

    // 2. Adicionar ao buffer
    this.buffer.push(entry);
    if (this.buffer.length > this.maxBufferSize) {
      this.buffer.shift();
    }

    // 3. Enviar logs críticos imediatamente
    if (level >= LogLevel.ERROR) {
      this.flushLogs();
    }
  }

  debug(message: string, context?: any) {
    this.log(LogLevel.DEBUG, message, context);
  }

  info(message: string, context?: any) {
    this.log(LogLevel.INFO, message, context);
  }

  warn(message: string, context?: any) {
    this.log(LogLevel.WARN, message, context);
  }

  error(message: string, context?: any) {
    this.log(LogLevel.ERROR, message, context);
  }

  // 4. Flush logs para servidor
  private async flushLogs() {
    if (this.buffer.length === 0) return;

    try {
      await fetch('/api/logs', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ logs: [...this.buffer] })
      });

      this.buffer = [];
    } catch (error) {
      console.warn('Falha ao enviar logs para servidor:', error);
    }
  }

  private getCurrentUserId(): number | undefined {
    // Implementar baseado no contexto de auth
    return undefined;
  }

  private getSessionId(): string | undefined {
    // Implementar baseado no sistema de sessão
    return undefined;
  }

  // 5. Método para download de logs local
  downloadLogs() {
    const logsJson = JSON.stringify(this.buffer, null, 2);
    const blob = new Blob([logsJson], { type: 'application/json' });
    const url = URL.createObjectURL(blob);

    const a = document.createElement('a');
    a.href = url;
    a.download = `logs-${new Date().toISOString().split('T')[0]}.json`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }
}

export const logger = new Logger(
