## 1. VISÃO GERAL DA ARQUITETURA
================================

### Tipo de Aplicação
- **Stack**: Full-stack web application (NÃO serverless)
- **Backend**: Express.js + SQLite + Drizzle ORM
- **Frontend**: React 18 + TypeScript + Vite
- **Deployment**: Docker container em servidor local
- **Banco de Dados**: SQLite com persistência em volume Docker
- **Autenticação**: Passport.js com estratégia local e sessões
- **Real-time**: WebSocket para atualizações em tempo real

### Características Principais
- Sistema de gestão de turnos para restaurante
- Controle de caixa com múltiplas formas de pagamento
- Gestão de estoque e produtos
- Sistema de folha de pagamento
- Relatórios semanais automatizados
- Controle de colaboradores e consumo interno
- Sistema de correções administrativas
- Geração de relatórios em PDF e Excel

```typescript
const calculateDivergence = () => {
  const initialCash = parseFloat(shift.initialCash || '0');
  const initialCoins = parseFloat(shift.initialCoins || '0');
  const cashSales = payments.cash || 0;

  // Valores esperados
  const expectedCash = initialCash + cashSales;
  const expectedCoins = initialCoins;

  // Calcular ajustes administrativos
  let totalAdjustments = 0;
  if (cashAdjustments && Array.isArray(cashAdjustments)) {
    totalAdjustments = cashAdjustments.reduce((total, adjustment) => {
      if (adjustment.type === 'withdraw' || adjustment.type === 'adjustment') {
        return total + parseFloat(adjustment.amount || '0');
      }
      return total;
    }, 0);
  }

  // Divergência final
  const totalExpected = expectedCash + expectedCoins;
  const totalCounted = finalCash + finalCoins;
  const baseDivergence = totalCounted - totalExpected;

  return baseDivergence + totalAdjustments;
};

// Calcula valores das maquininhas
```

### 4.3 Backend - Processador de Turnos

#### `shift-processor.ts`
**Localização**: `/server/lib/shift-processor.ts`

**Função Principal**: `getProcessedShiftById(shiftId: number)`

**Responsabilidades**:
- Buscar turno com todas as relações
- Aplicar correções administrativas
- Calcular valores derivados
- Garantir integridade dos dados

```typescript
export async function getProcessedShiftById(shiftId: number) {
  try {
    // 1. Buscar turno básico
    const shiftResults = await db
      .select()
      .from(shifts)
      .where(eq(shifts.id, shiftId))
      .limit(1);

    if (shiftResults.length === 0) return null;
    const shift = shiftResults[0];

    // 2. Buscar relações manualmente (evita problemas do Drizzle)
    const [user, records, payments, collaborators] = await Promise.all([
      db.select().from(users).where(eq(users.id, shift.userId)).limit(1),
      db.select({
        id: shiftRecords.id,
        shiftId: shiftRecords.shiftId,
        productId: shiftRecords.productId,
        entryQty: shiftRecords.entryQty,
        arrivalQty: shiftRecords.arrivalQty,
        leftoverQty: shiftRecords.leftoverQty,
        discardQty: shiftRecords.discardQty,
        consumedQty: shiftRecords.consumedQty,
        soldQty: shiftRecords.soldQty,
        productName: products.name,
        productPrice: products.price,
        productCategory: products.category
      })
      .from(shiftRecords)
      .leftJoin(products, eq(shiftRecords.productId, products.id))
      .where(eq(shiftRecords.shiftId, shiftId)),
      db.select().from(shiftPayments).where(eq(shiftPayments.shiftId, shiftId)).limit(1),
      db.select()
        .from(shiftCollaborators)
        .leftJoin(users, eq(shiftCollaborators.userId, users.id))
        .where(eq(shiftCollaborators.shiftId, shiftId))
    ]);

    // 3. Reconstruir objeto com relações
    const shiftWithRelations = {
      ...shift,
      user: user[0] || null,
      records: records.map(r => ({
        id: r.id,
        shiftId: r.shiftId,
        productId: r.productId,
        entryQty: r.entryQty,
        arrivalQty: r.arrivalQty,
        leftoverQty: r.leftoverQty,
        discardQty: r.discardQty,
        consumedQty: r.consumedQty,
        soldQty: r.soldQty,
        product: {
          name: r.productName,
          price: r.productPrice,
          category: r.productCategory
        }
      })),
      payments: payments[0] || null,
      collaborators: collaborators.map(c => ({
        id: c.shift_collaborators.id,
        user: c.users
      }))
    };

    // 4. Criar cópia profunda para evitar mutações
    const processedShift = JSON.parse(JSON.stringify(shiftWithRelations));

    // 5. Garantir estruturas padrão
    if (!processedShift.records) processedShift.records = [];
    if (!processedShift.payments) {
      processedShift.payments = {
        cash: "0.00", pix: "0.00", stoneCard: "0.00",
        stoneVoucher: "0.00", pagBankCard: "0.00"
      };
    }
    if (!processedShift.collaborators) processedShift.collaborators = [];

    // 6. Buscar e aplicar correções ativas
    const activeCorrections = await db
      .select()
      .from(corrections)
      .where(and(
        eq(corrections.shiftId, shiftId),
        isNull(corrections.revokedAt)
      ));

    // 7. Aplicar cada correção de forma defensiva
    for (const correction of activeCorrections) {
      try {
        switch (correction.correctionType) {
          case 'product_qty':
            if (correction.shiftRecordId && processedShift.records && correction.fieldName) {
              const recordIndex = processedShift.records.findIndex(
                r => r.id === correction.shiftRecordId
              );
              if (recordIndex !== -1) {
                const record = processedShift.records[recordIndex];
                record[correction.fieldName] = parseInt(correction.correctedValue) || 0;
              }
            }
            break;

          case 'product_price':
            if (correction.productId && processedShift.records) {
              for (const record of processedShift.records) {
                if (record.productId === correction.productId && record.product) {
                  record.product.price = correction.correctedValue || record.product.price;
                }
              }
            }
            break;

          case 'payment':
            if (correction.paymentMethod && processedShift.payments) {
              processedShift.payments[correction.paymentMethod] = correction.correctedValue || "0.00";
            }
            break;

          case 'cash_count':
            if (correction.cashType === 'countedFinalCash') {
              processedShift.countedFinalCash = correction.correctedValue;
              processedShift.finalCash = correction.correctedValue;
            } else if (correction.cashType === 'countedFinalCoins') {
              processedShift.countedFinalCoins = correction.correctedValue;
              processedShift.finalCoins = correction.correctedValue;
            }
            break;
        }
      } catch (correctionError) {
        console.error(`Erro ao aplicar correção ${correction.id}:`, correctionError);
      }
    }

    // 8. Buscar ajustes de caixa (sangrias)
    const shiftAdjustments = await db
      .select()
      .from(cashAdjustments)
      .where(eq(cashAdjustments.shiftId, shiftId));

    processedShift.cashAdjustments = shiftAdjustments || [];

    // 9. Recalcular valores derivados
    processedShift.totalCorrections = activeCorrections.length;

    // Calcular divergência de caixa
    let cashSales = 0;
    if (processedShift.payments && processedShift.payments.cash) {
      cashSales = parseFloat(processedShift.payments.cash) || 0;
    }

    const initialCash = parseFloat(processedShift.initialCash) || 0;
    const finalCash = parseFloat(processedShift.finalCash) || 0;
    processedShift.cashDivergence = (finalCash - (initialCash + cashSales)).toFixed(2);

    // Calcular total de sangrias
    let totalWithdrawals = 0;
    if (shiftAdjustments && Array.isArray(shiftAdjustments)) {
      totalWithdrawals = shiftAdjustments
        .filter(adj => adj.type === 'withdraw' || adj.type === 'sangria')
        .reduce((sum, adj) => sum + (parseFloat(adj.amount) || 0), 0);
    }
    processedShift.totalWithdrawals = totalWithdrawals.toFixed(2);

    // Calcular valores para próximo turno
    if (processedShift.cashForNextShift) {
      processedShift.cashForNextShift = processedShift.cashForNextShift;
    } else {
      const cashForNext = finalCash - totalWithdrawals;
      processedShift.cashForNextShift = cashForNext.toFixed(2);
    }

    if (processedShift.coinsForNextShift) {
      processedShift.coinsForNextShift = processedShift.coinsForNextShift;
    } else {
      processedShift.coinsForNextShift = processedShift.finalCoins || "0.00";
    }

    // Calcular total de vendas
    let totalSales = 0;
    if (processedShift.payments) {
      totalSales = Object.values(processedShift.payments).reduce((sum, value) => {
        return sum + (parseFloat(value) || 0);
      }, 0);
    }
    processedShift.totalSales = totalSales.toFixed(2);

    return processedShift;

  } catch (error) {
    console.error('Erro ao processar turno:', error);
    return null;
  }
}
```

```typescript
export function calculateCardPaymentValues(

  paymentType: 'stone_card' | 'stone_voucher' | 'pagbank_card'
): number {
  // Valor real = diferença entre acumulados
  const realValue = currentCumulative - lastShiftCumulative;

  // Aplicar validações específicas por tipo
  if (paymentType === 'stone_voucher') {
    // Vouchers têm comportamento especial
    return Math.max(0, realValue);
  }

  return Math.max(0, realValue);
}
```

### 4.4 Fórmulas de Negócio Principais

#### Fórmula Central de Estoque
**Localização**: `/client/src/lib/calculations.ts`

```
Quantidade Vendida = Entrada + Chegada - Sobra - Descarte - Consumo Interno
```

#### Divergência de Caixa
```
Divergência = Caixa Final - (Caixa Inicial + Vendas em Dinheiro - Sangrias)
```

#### Caixa para Próximo Turno
```
Caixa Próximo Turno = Caixa Final - Envelope Administrativo - Sangrias
```

#### Inconsistência de Vendas
```
Inconsistência = |Total Pagamentos - Total Vendas por Produtos|
```

### 4.5 Validações e Regras de Negócio

#### Divergência Máxima Permitida
**Constante**: `MAX_CASH_DIVERGENCE = 5.00` (R$ 5,00)

#### Valores Mínimos Recomendados

#### Observações Obrigatórias
- Divergência de caixa > R$ 5,00

### 7.1 Autenticação e Autorização

#### `use-auth.tsx`
**Hook principal de autenticação**:
```typescript
export function useAuth() {
  const { data: user, error, isLoading } = useQuery<User>({
    queryKey: ["/api/user"],
    staleTime: Infinity,
    retry: false,
  });

  const login = async (email: string, password: string) => {
    const response = await fetch("/api/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email, password }),
      credentials: "include",
    });

    if (response.ok) {
      queryClient.invalidateQueries({ queryKey: ["/api/user"] });
      return { success: true };
    }
    // Handle error
  };

  const logout = async () => {
    await fetch("/api/logout", {
      method: "POST",
      credentials: "include"
    });
    queryClient.clear();
  };

  return { user, login, logout, isLoading };
}
```

#### `protected-route.tsx`
**Proteção de rotas por role**:
```typescript
interface ProtectedRouteProps {
  children: React.ReactNode;
  requireAdmin?: boolean;
}

export function ProtectedRoute({ children, requireAdmin = false }: ProtectedRouteProps) {
  const { user, isLoading } = useAuth();

  if (isLoading) return <div>Carregando...</div>;

  if (!user) {
    return <Redirect to="/auth" />;
  }

  if (requireAdmin && user.role !== "admin") {
    return <div>Acesso negado. Apenas administradores.</div>;
  }

  return <>{children}</>;
}
```

```typescript
export const queryClient = new QueryClient({

export async function apiRequest(
  method: string,
  url: string,
  data?: any,
): Promise<Response> {
  const options: RequestInit = {
    method,
    headers: { "Content-Type": "application/json" },
    credentials: "include", // CRÍTICO para sessões
  };

  if (data) {
    options.body = JSON.stringify(data);
  }

  const response = await fetch(url, options);

  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }

  return response;
}
```

### 7.3 Real-time com WebSocket

#### `ws.ts` (Server)
```typescript
export function setupWebSocket(server: any) {
  const wss = new WebSocketServer({
    server,
    path: '/ws-api'
  });

  wss.on('connection', (ws, req) => {
    const url = new URL(req.url!, `http://${req.headers.host}`);
    const shiftId = url.searchParams.get('shiftId');

    if (shiftId) {
      ws.shiftId = shiftId;
      console.log(`WebSocket conectado para turno ${shiftId}`);
    }

    ws.on('close', () => {
      console.log(`WebSocket desconectado`);
    });
  });

  return wss;
}

export function notifyShiftClients(wss: WebSocketServer, shiftId: string, data: any) {
  wss.clients.forEach((client) => {
    if (client.readyState === WebSocket.OPEN && (client as any).shiftId === shiftId) {
      client.send(JSON.stringify({
        type: 'shift-updated',
        data
      }));
    }
  });
}
```
