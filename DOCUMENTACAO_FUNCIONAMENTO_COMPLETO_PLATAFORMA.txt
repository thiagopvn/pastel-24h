# DOCUMENTAÇÃO COMPLETA - FUNCIONAMENTO DA PLATAFORMA PASTEL24H

## VISÃO GERAL DA PLATAFORMA

O Pastel24h é uma aplicação full-stack tradicional (não serverless) para gerenciamento de turnos de restaurante, construída com Express.js no backend e React no frontend, utilizando SQLite como banco de dados e WebSocket para atualizações em tempo real.

## ARQUITETURA GERAL

### Stack Tecnológico
- **Backend**: Express.js + TypeScript + SQLite + Drizzle ORM + WebSocket
- **Frontend**: React 18 + TypeScript + Wouter + TanStack Query + Tailwind CSS
- **Banco de Dados**: SQLite com Drizzle ORM para queries type-safe
- **Autenticação**: Passport.js com estratégia local e sessões
- **Deploy**: Docker container em servidor local (não cloud/serverless)

```typescript
// server/auth.ts
import passport from 'passport';
import { Strategy as LocalStrategy } from 'passport-local';
import bcrypt from 'bcryptjs';

// Configuração da estratégia local
passport.use(new LocalStrategy({
    usernameField: 'email',
    passwordField: 'password'
  }, async (email, password, done) => {
    try {
      const user = await db.query.users.findFirst({
        where: eq(users.email, email)
      });

      if (!user) {
        return done(null, false, { message: 'Email não encontrado' });
      }

      // Verificar senha com bcrypt
      const isValidPassword = await bcrypt.compare(password, user.password);

      if (!isValidPassword) {
        return done(null, false, { message: 'Senha incorreta' });
      }

      return done(null, user);
    } catch (error) {
      return done(error);
    }
  }
));

// Serialização para sessão
passport.serializeUser((user: any, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id: number, done) => {
  try {
    const user = await db.query.users.findFirst({
      where: eq(users.id, id)
    });
    done(null, user);
  } catch (error) {
    done(error);
  }
});

// Middleware de autenticação
export const requireAuth = (req: any, res: any, next: any) => {
  if (req.isAuthenticated()) {
    return next();
  }
  res.status(401).json({ error: 'Não autorizado' });
};

// Middleware de admin
export const requireAdmin = (req: any, res: any, next: any) => {
  if (req.isAuthenticated() && req.user.role === 'admin') {
    return next();
  }
  res.status(403).json({ error: 'Acesso negado - apenas administradores' });
};
```

### Abertura de Turno

```typescript
// server/routes.ts - Abertura de turno
app.post('/api/shifts', requireAuth, async (req, res) => {
  const { initialCash, initialCoins } = req.body;

  try {
    // Buscar último turno fechado para herança
    const lastClosedShift = await db.query.shifts.findFirst({
      where: eq(shifts.status, 'closed'),
      orderBy: [desc(shifts.id)],
      with: {
        records: {
          with: { product: true }
        }
      }
    });

    // Calcular valores esperados baseados no turno anterior
    let expectedCash = "200.00";
    let expectedCoins = "50.00";

    if (lastClosedShift) {
      expectedCash = lastClosedShift.cashForNextShift || lastClosedShift.finalCash || "200.00";
      expectedCoins = lastClosedShift.coinsForNextShift || lastClosedShift.finalCoins || "50.00";
    }

    const informedCash = parseFloat(initialCash) || 0;
    const informedCoins = parseFloat(initialCoins) || 0;
    const expectedTotal = parseFloat(expectedCash) + parseFloat(expectedCoins);
    const informedTotal = informedCash + informedCoins;

    // Calcular divergência na abertura
    const openingDiscrepancy = informedTotal - expectedTotal;

    // Criar novo turno
    const [newShift] = await db.insert(shifts).values({
      userId: req.user.id,
      startTime: new Date().toISOString(),
      initialCash: informedCash.toFixed(2),
      initialCoins: informedCoins.toFixed(2),
      expectedCash: expectedCash,
      expectedCoins: expectedCoins,
      openingDiscrepancy: openingDiscrepancy.toFixed(2),
      status: 'open'
    }).returning();

    // Criar registro de pagamentos zerado
    await db.insert(shiftPayments).values({
      shiftId: newShift.id,
      cash: '0',
      pix: '0',
      stoneCard: '0',
      stoneVoucher: '0',
      pagBankCard: '0'
    });

    // Herdar produtos do turno anterior (sobras viram entradas)
    if (lastClosedShift?.records) {
      const inheritanceRecords = lastClosedShift.records
        .filter(record => parseFloat(record.leftoverQty || '0') > 0)
        .map(record => ({
          shiftId: newShift.id,
          productId: record.productId,
          entryQty: record.leftoverQty,  // Sobra do turno anterior
          arrivalQty: "0",
          discardQty: "0",
          consumptionQty: "0",
          leftoverQty: record.leftoverQty // Inicialmente igual à entrada
        }));

      if (inheritanceRecords.length > 0) {
        await db.insert(shiftRecords).values(inheritanceRecords);
      }
    }

    // Timeline de auditoria
    await db.insert(timeline).values({
      shiftId: newShift.id,
      userId: req.user.id,
      action: 'SHIFT_OPENED',
      description: `Turno aberto - Caixa: R$${informedCash.toFixed(2)}, Divergência: R$${openingDiscrepancy.toFixed(2)}`,
      metadata: JSON.stringify({
        expectedCash,
        expectedCoins,
        informedCash,
        informedCoins,
        openingDiscrepancy
      }),
      createdAt: new Date().toISOString()
    });

    res.status(201).json(newShift);
  } catch (error) {
    console.error('Erro ao abrir turno:', error);
    res.status(500).json({ error: 'Erro interno do servidor' });
  }
});
```

### Fechamento de Turno

```typescript
// server/routes.ts - Fechamento de turno
app.post('/api/shifts/:id/close', requireAuth, async (req, res) => {
  const shiftId = parseInt(req.params.id);
  const {
    finalCash,
    finalCoins,
    envelopeCash,
    envelopeCoins,
    payments,
    cumulativeCardValues
  } = req.body;

  try {
    // Verificar se turno existe e está aberto
    const shift = await db.query.shifts.findFirst({
      where: and(eq(shifts.id, shiftId), eq(shifts.status, 'open')),
      with: { user: true }
    });

    if (!shift) {
      return res.status(404).json({ error: 'Turno não encontrado ou já fechado' });
    }

    // Verificar autorização
    if (req.user.role !== 'admin' && req.user.id !== shift.userId) {
      return res.status(403).json({ error: 'Sem permissão para fechar este turno' });
    }

    let finalPayments = payments;

    // Calcular valores reais dos cartões se fornecidos valores cumulativos
    if (cumulativeCardValues) {
      const cardCalculator = new CardPaymentCalculator(db);
      const calculatedCards = await cardCalculator.calculateRealCardPayments(
        shiftId,
        { ...payments, ...cumulativeCardValues }
      );

      finalPayments = {
        cash: payments.cash,
        pix: payments.pix,
        stoneCard: calculatedCards.realValues.stoneCard.toFixed(2),
        stoneVoucher: calculatedCards.realValues.stoneVoucher.toFixed(2),
        pagBankCard: calculatedCards.realValues.pagBankCard.toFixed(2)
      };

      // Atualizar pagamentos com valores cumulativos
      await db.update(shiftPayments)
        .set({
          ...finalPayments,
          stoneCardCumulative: cumulativeCardValues.stoneCardCumulative,
          stoneVoucherCumulative: cumulativeCardValues.stoneVoucherCumulative,
          pagBankCardCumulative: cumulativeCardValues.pagBankCardCumulative,
          calculatedFromCumulative: true
        })
        .where(eq(shiftPayments.shiftId, shiftId));
    } else {
      // Atualizar apenas valores reais
      await db.update(shiftPayments)
        .set(finalPayments)
        .where(eq(shiftPayments.shiftId, shiftId));
    }

    // Calcular valores para próximo turno
    const actualCash = parseFloat(finalCash) || 0;
    const actualCoins = parseFloat(finalCoins) || 0;
    const envelopeCashAmount = parseFloat(envelopeCash) || 0;
    const envelopeCoinsAmount = parseFloat(envelopeCoins) || 0;

    const cashForNext = actualCash - envelopeCashAmount;
    const coinsForNext = actualCoins - envelopeCoinsAmount;

    // Fechar turno
    await db.update(shifts).set({
      status: 'closed',
      endTime: new Date().toISOString(),
      finalCash: actualCash.toFixed(2),
      finalCoins: actualCoins.toFixed(2),
      envelopeCash: envelopeCashAmount.toFixed(2),
      envelopeCoins: envelopeCoinsAmount.toFixed(2),
      cashForNextShift: cashForNext.toFixed(2),
      coinsForNextShift: coinsForNext.toFixed(2)
    }).where(eq(shifts.id, shiftId));

    // Calcular horas trabalhadas para colaboradores
    const collaborators = await db.query.shiftCollaborators.findMany({
      where: eq(shiftCollaborators.shiftId, shiftId)
    });

    for (const collaborator of collaborators) {
      if (!collaborator.endTime) {
        const startTime = new Date(collaborator.startTime);
        const endTime = new Date();
        const hoursWorked = (endTime.getTime() - startTime.getTime()) / (1000 * 60 * 60);

        await db.update(shiftCollaborators)
          .set({
            endTime: endTime.toISOString(),
            hoursWorked: Math.round(hoursWorked * 100) / 100
          })
          .where(eq(shiftCollaborators.id, collaborator.id));
      }
    }

    // Timeline de auditoria
    await db.insert(timeline).values({
      shiftId,
      userId: req.user.id,
      action: 'SHIFT_CLOSED',
      description: `Turno fechado - Total: R$${(actualCash + actualCoins).toFixed(2)}`,
      metadata: JSON.stringify({
        finalCash: actualCash,
        finalCoins: actualCoins,
        envelopeCash: envelopeCashAmount,
        envelopeCoins: envelopeCoinsAmount,
        payments: finalPayments
      }),
      createdAt: new Date().toISOString()
    });

    // Notificar clientes WebSocket
    notifyShiftClients(shiftId, 'shift-closed');

    res.json({ message: 'Turno fechado com sucesso' });
  } catch (error) {
    console.error('Erro ao fechar turno:', error);
    res.status(500).json({ error: 'Erro interno do servidor' });
  }
});
```

## SISTEMA DE INVENTÁRIO

### Fórmula Principal

```typescript
// client/src/lib/calculations.ts
export const INVENTORY_FORMULA = {
  // Quantidade Vendida = Entrada + Chegada - Sobra - Descarte - Consumo Interno
  soldQuantity: (entry: number, arrival: number, leftover: number, discard: number, consumption: number): number => {
    return entry + arrival - leftover - discard - consumption;
  },

  // Validação de quantidades
  validateQuantities: (quantities: {
    entry: number;
    arrival: number;
    leftover: number;
    discard: number;
    consumption: number;
  }): { isValid: boolean; errors: string[] } => {
    const errors: string[] = [];

    // Todas as quantidades devem ser não-negativas
    Object.entries(quantities).forEach(([key, value]) => {
      if (value < 0) {
        errors.push(`${key} não pode ser negativo`);
      }
    });

    // Quantidade vendida não pode ser negativa
    const sold = INVENTORY_FORMULA.soldQuantity(
      quantities.entry,
      quantities.arrival,
      quantities.leftover,
      quantities.discard,
      quantities.consumption
    );

    if (sold < 0) {
      errors.push('Quantidade vendida não pode ser negativa');
    }

    return {
      isValid: errors.length === 0,
      errors
    };
  }
};
```

### Atualização de Inventário

```typescript
// server/routes.ts - Atualizar registro de produto
app.put('/api/shifts/:shiftId/records/:recordId', requireAuth, async (req, res) => {
  const { shiftId, recordId } = req.params;
  const { entryQty, arrivalQty, leftoverQty, discardQty, consumptionQty } = req.body;

  try {
    // Verificar se turno está aberto
    const shift = await db.query.shifts.findFirst({
      where: and(eq(shifts.id, parseInt(shiftId)), eq(shifts.status, 'open'))
    });

    if (!shift) {
      return res.status(404).json({ error: 'Turno não encontrado ou já fechado' });
    }

    // Validar quantidades
    const quantities = {
      entry: parseFloat(entryQty) || 0,
      arrival: parseFloat(arrivalQty) || 0,
      leftover: parseFloat(leftoverQty) || 0,
      discard: parseFloat(discardQty) || 0,
      consumption: parseFloat(consumptionQty) || 0
    };

    const validation = INVENTORY_FORMULA.validateQuantities(quantities);

    if (!validation.isValid) {
      return res.status(400).json({
        error: 'Quantidades inválidas',
        details: validation.errors
      });
    }

    // Atualizar registro
    await db.update(shiftRecords)
      .set({
        entryQty: quantities.entry.toString(),
        arrivalQty: quantities.arrival.toString(),
        leftoverQty: quantities.leftover.toString(),
        discardQty: quantities.discard.toString(),
        consumptionQty: quantities.consumption.toString()
      })
      .where(eq(shiftRecords.id, parseInt(recordId)));

    // Timeline de auditoria
    const soldQty = INVENTORY_FORMULA.soldQuantity(
      quantities.entry,
      quantities.arrival,
      quantities.leftover,
      quantities.discard,
      quantities.consumption
    );

    await db.insert(timeline).values({
      shiftId: parseInt(shiftId),
      userId: req.user.id,
      action: 'INVENTORY_UPDATED',
      description: `Inventário atualizado - Vendido: ${soldQty}`,
      metadata: JSON.stringify({
        recordId: parseInt(recordId),
        quantities,
        soldQty
      }),
      createdAt: new Date().toISOString()
    });

    // Notificar clientes WebSocket
    notifyShiftClients(parseInt(shiftId), 'inventory-updated');

    res.json({ message: 'Inventário atualizado com sucesso' });
  } catch (error) {
    console.error('Erro ao atualizar inventário:', error);
    res.status(500).json({ error: 'Erro interno do servidor' });
  }
});

    // Validar método de pagamento
    if (!Object.keys(PAYMENT_RATES).includes(method)) {
      return res.status(400).json({ error: 'Método de pagamento inválido' });
    }

    // Validar valor
    const paymentAmount = parseFloat(amount);
    if (paymentAmount <= 0) {
      return res.status(400).json({ error: 'Valor deve ser maior que zero' });
    }

    // Buscar pagamentos atuais
    const currentPayments = await db.query.shiftPayments.findFirst({
      where: eq(shiftPayments.shiftId, shiftId)
    });

    if (!currentPayments) {
      return res.status(404).json({ error: 'Registro de pagamentos não encontrado' });
    }

    // Atualizar valor do método específico
    const currentValue = parseFloat(currentPayments[method as keyof typeof currentPayments] as string || '0');
    const newValue = currentValue + paymentAmount;

    await db.update(shiftPayments)
      .set({ [method]: newValue.toFixed(2) })
      .where(eq(shiftPayments.shiftId, shiftId));

    // Timeline de auditoria
    await db.insert(timeline).values({
      shiftId,
      userId: req.user.id,
      action: 'PAYMENT_ADDED',
      description: `Pagamento ${PAYMENT_METHODS[method as keyof typeof PAYMENT_METHODS].label}: R$${paymentAmount.toFixed(2)}`,
      metadata: JSON.stringify({
        method,
        amount: paymentAmount,
        previousValue: currentValue,
        newValue
      }),
      createdAt: new Date().toISOString()
    });

    // Notificar clientes WebSocket
    notifyShiftClients(shiftId, 'payment-added');

    res.json({ message: 'Pagamento registrado com sucesso' });
  } catch (error) {
    console.error('Erro ao registrar pagamento:', error);
    res.status(500).json({ error: 'Erro interno do servidor' });
  }
});

// server/lib/shift-processor.ts
export const calculateCashDivergence = (shiftData: {
  initialCash: number;
  finalCash: number;
  cashSales: number;
  cashWithdrawals: number;
}): number => {
  // Fórmula: Divergência = Caixa Final - (Caixa Inicial + Vendas Cash - Sangrias)
  const expectedFinalCash = shiftData.initialCash + shiftData.cashSales - shiftData.cashWithdrawals;
  return shiftData.finalCash - expectedFinalCash;
};

// Exemplo de uso
const exampleCalculation = {
  initialCash: 200.00,
  finalCash: 450.00,
  cashSales: 300.00,
  cashWithdrawals: 50.00
};

// Expected: 200 + 300 - 50 = 450
// Actual: 450
// Divergence: 450 - 450 = 0 (sem divergência)
const divergence = calculateCashDivergence(exampleCalculation);

import { useEffect, useRef } from 'react';
import { useQueryClient } from '@tanstack/react-query';

export const useWebSocket = (shiftId: number | null) => {
  const wsRef = useRef<WebSocket | null>(null);
  const queryClient = useQueryClient();

  useEffect(() => {
    if (!shiftId) return;

    // Conectar ao WebSocket
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const wsUrl = `${protocol}//${window.location.host}/ws-api?shiftId=${shiftId}`;

    wsRef.current = new WebSocket(wsUrl);

    wsRef.current.onopen = () => {
      console.log('Conectado ao WebSocket');
    };

    wsRef.current.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);

        switch (data.type) {
          case 'shift-updated':
            // Invalidar queries relacionadas ao turno
            queryClient.invalidateQueries(['shift', shiftId]);
            queryClient.invalidateQueries(['shift-records', shiftId]);
            queryClient.invalidateQueries(['shift-payments', shiftId]);
            break;

          case 'pong':
            console.log('Pong recebido');
            break;
        }
      } catch (error) {
        console.error('Erro ao processar mensagem WebSocket:', error);
      }
    };

    wsRef.current.onclose = () => {
      console.log('Conexão WebSocket fechada');
    };

    wsRef.current.onerror = (error) => {
      console.error('Erro WebSocket:', error);
    };

    // Cleanup
    return () => {
      if (wsRef.current) {
        wsRef.current.close();
      }
    };
  }, [shiftId, queryClient]);

  // Função para enviar mensagens
  const sendMessage = (message: any) => {
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(message));
    }
  };

  return { sendMessage };
};

        sp.stone_card_cumulative,
        sp.stone_voucher_cumulative,

        sp.calculated_from_cumulative,

        -- Totais calculados
        (CAST(COALESCE(sp.cash, '0') AS REAL) +
         CAST(COALESCE(sp.pix, '0') AS REAL) +
         CAST(COALESCE(sp.stone_card, '0') AS REAL) +
         CAST(COALESCE(sp.stone_voucher, '0') AS REAL) +
         CAST(COALESCE(sp.pagbank_card, '0') AS REAL)) as total_payments,

        -- Cálculo de divergência de caixa
        (CAST(COALESCE(s.final_cash, '0') AS REAL) -
         (CAST(COALESCE(s.initial_cash, '0') AS REAL) +
          CAST(COALESCE(sp.cash, '0') AS REAL) -
          COALESCE((SELECT SUM(amount) FROM cash_adjustments
                   WHERE shift_id = s.id AND type = 'withdrawal'), 0))) as cash_divergence

      FROM shifts s
      LEFT JOIN users u ON s.user_id = u.id
      LEFT JOIN shift_payments sp ON s.id = sp.shift_id
      WHERE s.id = ${shiftId}
    `);

    if (!shiftData.rows.length) {
      return null;
    }

    const row = shiftData.rows[0] as any;

    // Buscar registros de produtos
    const recordsData = await db.execute(sql`
      SELECT
        sr.id,
        sr.product_id,
        sr.entry_qty,
        sr.arrival_qty,
        sr.leftover_qty,
        sr.discard_qty,
        sr.consumption_qty,

        -- Calcular quantidade vendida
        (CAST(COALESCE(sr.entry_qty, '0') AS REAL) +
         CAST(COALESCE(sr.arrival_qty, '0') AS REAL) -
         CAST(COALESCE(sr.leftover_qty, '0') AS REAL) -
         CAST(COALESCE(sr.discard_qty, '0') AS REAL) -
         CAST(COALESCE(sr.consumption_qty, '0') AS REAL)) as sold_qty,

        -- Calcular receita
        (CAST(COALESCE(sr.entry_qty, '0') AS REAL) +
         CAST(COALESCE(sr.arrival_qty, '0') AS REAL) -
         CAST(COALESCE(sr.leftover_qty, '0') AS REAL) -
         CAST(COALESCE(sr.discard_qty, '0') AS REAL) -
         CAST(COALESCE(sr.consumption_qty, '0') AS REAL)) * p.price as revenue,

        -- Dados do produto
        p.name as product_name,
        p.category as product_category,
        p.price as product_price

      FROM shift_records sr
      JOIN products p ON sr.product_id = p.id
      WHERE sr.shift_id = ${shiftId}
      ORDER BY p.category, p.name
    `);

    // Processar registros
    const records = recordsData.rows.map((record: any) => ({

      product: {
        id: record.product_id,
        name: record.product_name,
        category: record.product_category,
        price: record.product_price
      }
    }));

    // Calcular totais
    const totalRevenue = records.reduce((sum, record) => sum + (record.revenue || 0), 0);
    const totalSoldItems = records.reduce((sum, record) => sum + (record.soldQty || 0), 0);

    // Construir objeto processado
    const processedShift: ProcessedShift = {
      id: row.id,
      userId: row.user_id,
      startTime: row.start_time,
      endTime: row.end_time,
      status: row.status,

      finalCash: parseFloat(row.final_cash || '0'),

      finalCoins: parseFloat(row.final_coins || '0'),
      openingDiscrepancy: parseFloat(row.opening_discrepancy || '0'),
      cashDivergence: row.cash_divergence || 0,

      user: {
        id: row.user_id,
        name: row.user_name,

        stoneVoucher: parseFloat(row.stone_voucher || '0'),
        pagBankCard: parseFloat(row.pagbank_card || '0'),
        total: row.total_payments || 0
      },

      records,

      totals: {
        revenue: totalRevenue,
        soldItems: totalSoldItems,
        averageTicket: totalSoldItems > 0 ? totalRevenue / totalSoldItems : 0
      }
    };

    record.product.name,
    record.entryQty,
    record.arrivalQty,
    record.leftoverQty,
    record.discardQty,
    record.consumptionQty,
    record.soldQty?.toString() || '0',
    `R$ ${(record.revenue || 0).toFixed(2)}`
  ]);

  (doc as any).autoTable({
    startY: yPosition,
    head: [['Produto', 'Entrada', 'Chegada', 'Sobra', 'Descarte', 'Consumo', 'Vendido', 'Receita']],
    body: inventoryData,
    theme: 'grid',
    styles: { fontSize: 8 },
    headStyles: { fillColor: [66, 139, 202] }
  });

  // Salvar PDF
  const fileName = `turno_${shift.id}_${new Date().toISOString().split('T')[0]}.pdf`;
  doc.save(fileName);
};
```

import * as XLSX from 'xlsx';

export const exportShiftToExcel = (shift: ProcessedShift): void => {

## CONSIDERAÇÕES FINAIS

### Principais Características da Plataforma

1. **Arquitetura Tradicional**: Aplicação full-stack não-serverless com servidor Express.js persistente
2. **Banco de Dados Robusto**: SQLite com Drizzle ORM para type-safety e performance
3. **Tempo Real**: WebSocket para atualizações instantâneas entre clientes
4. **Autenticação Segura**: Passport.js com sessões e bcrypt para senhas
5. **Frontend Moderno**: React 18 com TypeScript, TanStack Query e Tailwind CSS
6. **Validações Rigorosas**: Zod para validação de dados e regras de negócio
7. **Auditoria Completa**: Timeline de ações para rastreabilidade
8. **Exportação Flexível**: PDF e Excel para relatórios
9. **Deploy Containerizado**: Docker para deploy em servidor local

### Fluxo de Dados Típico

1. **Frontend**: Usuário interage com componente React
2. **API Client**: Requisição HTTP com credenciais de sessão
3. **Backend**: Middleware de autenticação → Validação → Processamento → Banco de dados
4. **WebSocket**: Notificação em tempo real para clientes conectados
5. **Frontend**: Invalidação de cache e re-render automático

Esta documentação apresenta uma visão completa do funcionamento da plataforma Pastel24h, desde a arquitetura de alto nível até exemplos detalhados de código, fornecendo uma base sólida para desenvolvimento e manutenção do sistema.
