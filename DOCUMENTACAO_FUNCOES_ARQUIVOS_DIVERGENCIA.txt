// ANÁLISE: Núcleo dos cálculos de divergência de caixa

export class DivergenceCalculator {
  constructor(private db: any) {}

  // FUNÇÃO PRINCIPAL: Calcular divergência de um turno
  async calculateShiftDivergence(shiftId: number): Promise<DivergenceResult> {
    // 1. BUSCAR DADOS FUNDAMENTAIS
    const shiftData = await this.getShiftFinancialData(shiftId);

    // 2. APLICAR FÓRMULA DE DIVERGÊNCIA
    const divergence = this.applyDivergenceFormula(shiftData);

    // 3. ANALISAR PADRÕES E CAUSAS
    const analysis = await this.analyzeDivergencePatterns(shiftId, divergence);

    return {
      shiftId,
      divergence,
      analysis,
      calculatedAt: new Date().toISOString()
    };
  }

  // FUNÇÃO CORE: Aplicar fórmula matemática
  private applyDivergenceFormula(data: ShiftFinancialData): number {
    /*
    FÓRMULA FUNDAMENTAL:
    Divergência = Caixa Final Contado - Caixa Esperado

    Onde:
    Caixa Esperado = Caixa Inicial + Vendas em Dinheiro - Sangrias

    EXEMPLO:
    - Caixa Inicial: R$ 200,00
    - Vendas Dinheiro: R$ 300,00
    - Sangrias: R$ 50,00
    - Caixa Final: R$ 460,00

    Esperado = 200 + 300 - 50 = 450
    Divergência = 460 - 450 = +10 (sobra de R$ 10)
    */

    const expectedCash = data.initialCash + data.cashSales - data.cashWithdrawals;
    const divergence = data.finalCash - expectedCash;

    console.log(`🧮 Calculando divergência para turno ${data.shiftId}:`);
    console.log(`   💰 Inicial: R$ ${data.initialCash.toFixed(2)}`);
    console.log(`   📈 Vendas: R$ ${data.cashSales.toFixed(2)}`);
    console.log(`   📤 Sangrias: R$ ${data.cashWithdrawals.toFixed(2)}`);
    console.log(`   🎯 Esperado: R$ ${expectedCash.toFixed(2)}`);
    console.log(`   📊 Final: R$ ${data.finalCash.toFixed(2)}`);
    console.log(`   ⚖️  Divergência: R$ ${divergence.toFixed(2)}`);

    return divergence;
  }

  // FUNÇÃO: Buscar dados financeiros completos
  private async getShiftFinancialData(shiftId: number): Promise<ShiftFinancialData> {
    // Query otimizada para buscar todos os dados necessários
    const result = await this.db.execute(sql`
      SELECT
        s.id,
        s.initial_cash,
        s.final_cash,
        sp.cash as cash_sales,
        COALESCE(
          (SELECT SUM(amount) FROM cash_adjustments
           WHERE shift_id = s.id AND type = 'withdrawal'),
          0
        ) as cash_withdrawals
      FROM shifts s
      LEFT JOIN shift_payments sp ON s.id = sp.shift_id
      WHERE s.id = ${shiftId}
    `);

    if (!result.rows.length) {
      throw new Error(`Turno ${shiftId} não encontrado`);
    }

    const row = result.rows[0];
    return {
      shiftId,
      initialCash: parseFloat(row.initial_cash || '0'),
      finalCash: parseFloat(row.final_cash || '0'),
      cashSales: parseFloat(row.cash_sales || '0'),
      cashWithdrawals: parseFloat(row.cash_withdrawals || '0')
    };
  }

  // FUNÇÃO: Analisar padrões de divergência
  private async analyzeDivergencePatterns(
    shiftId: number,
    divergence: number
  ): Promise<DivergenceAnalysis> {

    // Classificar severidade
    const severity = this.classifyDivergenceSeverity(Math.abs(divergence));

    // Identificar tipo
    const type = divergence > 0 ? 'sobra' : divergence < 0 ? 'falta' : 'balanceado';

    // Buscar padrões históricos do funcionário
    const historicalPattern = await this.getEmployeeHistoricalPattern(shiftId);

    // Gerar causas possíveis baseadas na análise
    const possibleCauses = this.identifyPossibleCauses(divergence, historicalPattern);

    // Gerar recomendações
    const recommendations = this.generateRecommendations(divergence, severity, historicalPattern);

    return {
      amount: divergence,
      type,
      severity,
      possibleCauses,
      recommendations,
      historicalContext: historicalPattern
    };
  }

  // FUNÇÃO: Classificar severidade da divergência
  private classifyDivergenceSeverity(absDivergence: number): 'low' | 'medium' | 'high' {
    /*
    CRITÉRIOS DE SEVERIDADE:
    - Baixa: até R$ 5,00 (variação normal de contagem)
    - Média: R$ 5,01 a R$ 20,00 (requer atenção)
    - Alta: acima de R$ 20,00 (requer investigação)
    */
    if (absDivergence <= 5) return 'low';
    if (absDivergence <= 20) return 'medium';
    return 'high';
  }

  // FUNÇÃO: Identificar causas possíveis
  private identifyPossibleCauses(
    divergence: number,
    historicalPattern: HistoricalPattern
  ): string[] {
    const causes: string[] = [];

    if (divergence > 0) {
      // SOBRAS - possíveis causas
      causes.push('Erro na contagem de caixa final');
      causes.push('Venda não registrada no sistema');
      causes.push('Troco dado em valor menor que registrado');

      if (historicalPattern.averageDivergence > 0) {
        causes.push('Padrão recorrente de sobras - revisar procedimentos');
      }
    } else if (divergence < 0) {
      // FALTAS - possíveis causas
      causes.push('Sangria não registrada no sistema');
      causes.push('Erro na contagem de caixa final');
      causes.push('Troco dado em valor maior que registrado');
      causes.push('Possível furto ou perda');

      if (historicalPattern.averageDivergence < 0) {
        causes.push('Padrão recorrente de faltas - investigação necessária');
      }
    }

    // Causas baseadas em magnitude
    if (Math.abs(divergence) > 50) {
      causes.push('Divergência alta - possível erro sistêmico');
      causes.push('Verificar se todas as sangrias foram registradas');
    }

    return causes;
  }

  // FUNÇÃO: Gerar recomendações
  private generateRecommendations(
    divergence: number,
    severity: 'low' | 'medium' | 'high',
    historicalPattern: HistoricalPattern
  ): string[] {
    const recommendations: string[] = [];

    switch (severity) {
      case 'low':
        recommendations.push('Verificar contagem de moedas');
        recommendations.push('Revisar cálculos de troco');
        break;

      case 'medium':
        recommendations.push('Recontagem obrigatória do caixa');
        recommendations.push('Verificar registros de sangria');
        recommendations.push('Revisar vendas do período');
        break;

      case 'high':
        recommendations.push('Investigação imediata necessária');
        recommendations.push('Revisar todo o fluxo de caixa do turno');
        recommendations.push('Verificar câmeras de segurança');
        recommendations.push('Auditoria completa dos registros');
        break;
    }

    // Recomendações baseadas em padrões
    if (historicalPattern.divergenceRate > 0.3) {
      recommendations.push('Treinamento adicional recomendado');
      recommendations.push('Supervisão mais próxima nos próximos turnos');
    }

    return recommendations;
  }
}

#### `server/lib/shift-inheritance.ts` - Herança Entre Turnos

```typescript
// ANÁLISE: Gerencia a transição de valores entre turnos consecutivos

export class ShiftInheritanceService {
  constructor(private db: any) {}

  // FUNÇÃO PRINCIPAL: Calcular herança para novo turno
  async calculateInheritanceValues(lastClosedShiftId?: number): Promise<InheritanceData> {
    /*
    LÓGICA DE HERANÇA:
    1. Buscar último turno fechado
    2. Calcular valores disponíveis para próximo turno
    3. Considerar envelope (dinheiro retirado)
    4. Herdar produtos em estoque
    */

    const lastShift = await this.getLastClosedShift(lastClosedShiftId);

    if (!lastShift) {
      return this.getDefaultInheritanceValues();
    }

    // CÁLCULO DE HERANÇA DE DINHEIRO
    const cashInheritance = this.calculateCashInheritance(lastShift);

    // HERANÇA DE PRODUTOS
    const productInheritance = this.calculateProductInheritance(lastShift.records);

    return {
      cash: cashInheritance,
      products: productInheritance,
      source: {
        shiftId: lastShift.id,
        closedAt: lastShift.endTime,
        userId: lastShift.userId
      }
    };
  }

  // FUNÇÃO: Calcular herança de dinheiro
  private calculateCashInheritance(lastShift: any): CashInheritanceData {
    /*
    FÓRMULA DE HERANÇA:
    Dinheiro para próximo turno = Caixa Final - Envelope

    EXEMPLO:
    - Caixa Final: R$ 500,00
    - Envelope: R$ 200,00
    - Para próximo: R$ 300,00
    */

    const finalCash = parseFloat(lastShift.finalCash || '0');
    const finalCoins = parseFloat(lastShift.finalCoins || '0');
    const envelopeCash = parseFloat(lastShift.envelopeCash || '0');
    const envelopeCoins = parseFloat(lastShift.envelopeCoins || '0');

    // Valores específicos calculados no fechamento
    const cashForNext = parseFloat(lastShift.cashForNextShift || '0');
    const coinsForNext = parseFloat(lastShift.coinsForNextShift || '0');

    if (cashForNext > 0 || coinsForNext > 0) {
      // Usar valores específicos calculados
      return {
        expectedCash: cashForNext.toFixed(2),
        expectedCoins: coinsForNext.toFixed(2),
        source: 'calculated',
        calculation: {
          finalCash,
          finalCoins,
          envelopeCash,
          envelopeCoins,
          remaining: { cash: cashForNext, coins: coinsForNext }
        }
      };
    }

    // Fallback: calcular com base nos valores finais
    const inheritedCash = finalCash - envelopeCash;
    const inheritedCoins = finalCoins - envelopeCoins;

    return {
      expectedCash: Math.max(0, inheritedCash).toFixed(2),
      expectedCoins: Math.max(0, inheritedCoins).toFixed(2),
      source: 'fallback',
      calculation: {
        finalCash,
        finalCoins,
        envelopeCash,
        envelopeCoins,
        remaining: {
          cash: Math.max(0, inheritedCash),
          coins: Math.max(0, inheritedCoins)
        }
      }
    };
  }

  // FUNÇÃO: Validar transição entre turnos
  async validateShiftTransition(
    lastShiftId: number,
    newShiftData: { initialCash: number; initialCoins: number }
  ): Promise<TransitionValidationResult> {

    const inheritance = await this.calculateInheritanceValues(lastShiftId);

    const expectedCash = parseFloat(inheritance.cash.expectedCash);
    const expectedCoins = parseFloat(inheritance.cash.expectedCoins);

    const cashDiscrepancy = newShiftData.initialCash - expectedCash;
    const coinsDiscrepancy = newShiftData.initialCoins - expectedCoins;

    const validationResult: TransitionValidationResult = {
      isValid: true,
      discrepancies: [],
      warnings: [],
      inheritance
    };

    // Verificar discrepâncias significativas
    if (Math.abs(cashDiscrepancy) > 0.01) {
      validationResult.discrepancies.push({
        type: 'cash',
        expected: expectedCash,
        actual: newShiftData.initialCash,
        difference: cashDiscrepancy,
        severity: Math.abs(cashDiscrepancy) > 10 ? 'high' : 'medium'
      });
    }

    if (Math.abs(coinsDiscrepancy) > 0.01) {
      validationResult.discrepancies.push({
        type: 'coins',
        expected: expectedCoins,
        actual: newShiftData.initialCoins,
        difference: coinsDiscrepancy,
        severity: Math.abs(coinsDiscrepancy) > 5 ? 'high' : 'medium'
      });
    }

    // Marcar como inválido se há discrepâncias de alta severidade
    const hasHighSeverity = validationResult.discrepancies.some(d => d.severity === 'high');
    if (hasHighSeverity) {
      validationResult.isValid = false;
    }

    return validationResult;
  }
}

// ANÁLISE: Orquestrador principal do processamento de turnos

export class ShiftProcessor {
  constructor(private db: any) {
    this.divergenceCalculator = new DivergenceCalculator(db);
    this.inheritanceService = new ShiftInheritanceService(db);
  }

  // FUNÇÃO PRINCIPAL: Processar turno completo
  async processShift(shiftId: number): Promise<ProcessedShift> {
    /*
    FLUXO DE PROCESSAMENTO:
    1. Buscar dados básicos do turno
    2. Calcular divergência de caixa
    3. Processar registros de inventário
    4. Calcular totais financeiros
    5. Analisar padrões e tendências
    6. Gerar relatório consolidado
    */

    console.log(`🔄 Iniciando processamento do turno ${shiftId}`);

    // 1. DADOS BÁSICOS
    const shiftData = await this.getShiftBaseData(shiftId);

    // 2. CÁLCULO DE DIVERGÊNCIA
    const divergenceResult = await this.divergenceCalculator.calculateShiftDivergence(shiftId);

    // 3. PROCESSAMENTO DE INVENTÁRIO
    const inventoryData = await this.processInventoryRecords(shiftId);

    // 4. TOTAIS FINANCEIROS
    const financialTotals = await this.calculateFinancialTotals(shiftId);

    // 5. ANÁLISE DE PERFORMANCE
    const performanceAnalysis = await this.analyzeShiftPerformance(shiftId, {
      divergence: divergenceResult,
      inventory: inventoryData,
      financial: financialTotals
    });

    // 6. CONSOLIDAR RESULTADO
    const processedShift: ProcessedShift = {
      ...shiftData,
      divergenceAnalysis: divergenceResult,
      inventoryData,
      financialTotals,
      performanceAnalysis,
      processedAt: new Date().toISOString()
    };

    console.log(`✅ Turno ${shiftId} processado com sucesso`);

    return processedShift;
  }

  // FUNÇÃO: Calcular totais financeiros
  private async calculateFinancialTotals(shiftId: number): Promise<FinancialTotals> {
    const result = await this.db.execute(sql`
      SELECT
        -- PAGAMENTOS
        sp.cash,
        sp.pix,
        sp.stone_card,
        sp.stone_voucher,
        sp.pagbank_card,

        -- TOTAIS CALCULADOS
        (CAST(sp.cash AS REAL) + CAST(sp.pix AS REAL) +
         CAST(sp.stone_card AS REAL) + CAST(sp.stone_voucher AS REAL) +
         CAST(sp.pagbank_card AS REAL)) as total_payments,

        -- SANGRIAS
        COALESCE(
          (SELECT SUM(amount) FROM cash_adjustments
           WHERE shift_id = ${shiftId} AND type = 'withdrawal'),
          0
        ) as total_withdrawals,

        -- RECEITA DE PRODUTOS
        COALESCE(
          (SELECT SUM(
            (CAST(sr.entry_qty AS REAL) + CAST(sr.arrival_qty AS REAL) -
             CAST(sr.leftover_qty AS REAL) - CAST(sr.discard_qty AS REAL) -
             CAST(sr.consumption_qty AS REAL)) * p.price
          ) FROM shift_records sr
          JOIN products p ON sr.product_id = p.id
          WHERE sr.shift_id = ${shiftId}),
          0
        ) as total_revenue

      FROM shift_payments sp
      WHERE sp.shift_id = ${shiftId}
    `);

    const row = result.rows[0];

    return {
      payments: {
        cash: parseFloat(row.cash || '0'),
        pix: parseFloat(row.pix || '0'),
        stoneCard: parseFloat(row.stone_card || '0'),
        stoneVoucher: parseFloat(row.stone_voucher || '0'),
        pagBankCard: parseFloat(row.pagbank_card || '0'),
        total: parseFloat(row.total_payments || '0')
      },
      revenue: parseFloat(row.total_revenue || '0'),
      withdrawals: parseFloat(row.total_withdrawals || '0'),
      netCashFlow: parseFloat(row.cash || '0') - parseFloat(row.total_withdrawals || '0')
    };
  }

  // FUNÇÃO: Analisar performance do turno
  private async analyzeShiftPerformance(
    shiftId: number,
    data: { divergence: any; inventory: any; financial: any }
  ): Promise<PerformanceAnalysis> {

    // Métricas de qualidade
    const qualityMetrics = {
      divergenceScore: this.calculateDivergenceScore(data.divergence.amount),
      inventoryAccuracy: this.calculateInventoryAccuracy(data.inventory),
      financialConsistency: this.calculateFinancialConsistency(data.financial)
    };

    // Score geral (0-100)
    const overallScore = (
      qualityMetrics.divergenceScore * 0.4 +
      qualityMetrics.inventoryAccuracy * 0.3 +
      qualityMetrics.financialConsistency * 0.3
    );

    // Identificar áreas de melhoria
    const improvementAreas = [];
    if (qualityMetrics.divergenceScore < 80) {
      improvementAreas.push('Controle de caixa');
    }
    if (qualityMetrics.inventoryAccuracy < 90) {
      improvementAreas.push('Precisão do inventário');
    }
    if (qualityMetrics.financialConsistency < 85) {
      improvementAreas.push('Consistência financeira');
    }

    return {
      overallScore: Math.round(overallScore),
      qualityMetrics,
      improvementAreas,
      recommendations: this.generatePerformanceRecommendations(qualityMetrics)
    };
  }

  // FUNÇÃO: Calcular score de divergência (0-100)
  private calculateDivergenceScore(divergence: number): number {
    const absDivergence = Math.abs(divergence);

    // Score baseado na magnitude da divergência
    if (absDivergence === 0) return 100;
    if (absDivergence <= 2) return 95;
    if (absDivergence <= 5) return 85;
    if (absDivergence <= 10) return 70;
    if (absDivergence <= 20) return 50;
    if (absDivergence <= 50) return 25;
    return 0;
  }
}

  /*
  RESPONSABILIDADES:
  1. Validar permissões e estado do turno
  2. Processar pagamentos (incluindo cartões cumulativos)
  3. Calcular divergência de fechamento
  4. Determinar valores para próximo turno
  5. Registrar observações sobre divergências
  6. Notificar administradores
  7. Registrar auditoria completa

  const shiftId = parseInt(req.params.id);
  const {
    finalCash,
    finalCoins,
    envelopeCash,
    envelopeCoins,
    payments,
    cumulativeCardValues,
    observations
  } = req.body;

    await db.transaction(async (tx) => {
      // VALIDAÇÃO DO TURNO

        where: and(eq(shifts.id, shiftId), eq(shifts.status, 'open')),
        with: { user: true, payments: true }
      });

      if (!shift) {
        return res.status(404).json({ error: 'Turno não encontrado ou já fechado' });
      }

      // PROCESSAMENTO DE PAGAMENTOS
      let finalPayments = payments;
      if (cumulativeCardValues) {

        const calculatedCards = await cardCalculator.calculateRealCardPayments(

          { ...payments, ...cumulativeCardValues }
        );

        finalPayments = {
          cash: payments.cash,
          pix: payments.pix,
          stoneCard: calculatedCards.realValues.stoneCard.toFixed(2),
          stoneVoucher: calculatedCards.realValues.stoneVoucher.toFixed(2),
          pagBankCard: calculatedCards.realValues.pagBankCard.toFixed(2)
        };
      }

      // CÁLCULO DE DIVERGÊNCIA
      const divergenceCalculator = new DivergenceCalculator(tx);
      const shiftFinancialData = {
        shiftId,
        initialCash: parseFloat(shift.initialCash || '0'),
        finalCash: parseFloat(finalCash) || 0,
        cashSales: parseFloat(finalPayments.cash || '0'),
        cashWithdrawals: await this.getCashWithdrawals(tx, shiftId)
      };

      const divergence = divergenceCalculator.applyDivergenceFormula(shiftFinancialData);

      // VALORES PARA PRÓXIMO TURNO
      const actualCash = parseFloat(finalCash) || 0;
      const actualCoins = parseFloat(finalCoins) || 0;
      const envelopeCashAmount = parseFloat(envelopeCash) || 0;
      const envelopeCoinsAmount = parseFloat(envelopeCoins) || 0;

      const cashForNext = actualCash - envelopeCashAmount;
      const coinsForNext = actualCoins - envelopeCoinsAmount;

      // ATUALIZAR TURNO
      await tx.update(shifts).set({
        status: 'closed',
        endTime: new Date().toISOString(),
        finalCash: actualCash.toFixed(2),
        finalCoins: actualCoins.toFixed(2),

        closingObservations: observations || null

// ANÁLISE: Hook customizado para cálculos de divergência no frontend

export const useDivergenceCalculation = () => {
  /*
  RESPONSABILIDADES:
  1. Aplicar fórmula de divergência
  2. Classificar severidade
  3. Gerar sugestões de ação
  4. Memoizar resultados para performance
  */

  // FUNÇÃO PRINCIPAL: Calcular divergência
  const calculateDivergence = useCallback((data: {
    initialCash: number;
    finalCash: number;
    cashSales: number;
    cashWithdrawals: number;
  }) => {
    /*
    FÓRMULA REPLICADA DO BACKEND:
    expectedCash = initialCash + cashSales - cashWithdrawals
    divergence = finalCash - expectedCash
    */

    const expectedCash = data.initialCash + data.cashSales - data.cashWithdrawals;
    const divergence = data.finalCash - expectedCash;

    // Classificar severidade
    const severity = classifyDivergenceSeverity(Math.abs(divergence));

    // Gerar sugestões
    const suggestions = generateActionSuggestions(divergence, severity);

    return {
      expectedCash,
      divergence,
      severity,
      suggestions,
      isBalanced: Math.abs(divergence) <= 0.01
    };
  }, []);

  // FUNÇÃO: Classificar severidade
  const classifyDivergenceSeverity = (absDivergence: number): 'low' | 'medium' | 'high' => {
    if (absDivergence <= 5) return 'low';
    if (absDivergence <= 20) return 'medium';
    return 'high';
  };

  // FUNÇÃO: Gerar sugestões de ação
  const generateActionSuggestions = (divergence: number, severity: string): string[] => {
    const suggestions: string[] = [];

    if (Math.abs(divergence) <= 0.01) {
      suggestions.push('✅ Caixa balanceado - pode fechar o turno');
      return suggestions;
    }

    if (divergence > 0) {
      // SOBRAS
      suggestions.push('🔍 Verificar se todas as vendas foram registradas');
      suggestions.push('💰 Recontar o dinheiro em caixa');
      suggestions.push('📝 Verificar se houve trocos menores que o devido');
    } else {
      // FALTAS
      suggestions.push('📤 Verificar se todas as sangrias foram registradas');
      suggestions.push('💰 Recontar o dinheiro em caixa');
      suggestions.push('📝 Verificar se houve trocos maiores que o devido');
    }

    if (severity === 'high') {
      suggestions.push('🚨 Divergência alta - chamar supervisor');
      suggestions.push('📹 Verificar câmeras de segurança se disponível');
    }

    return suggestions;
  };

  // FUNÇÃO: Validar dados de entrada
  const validateCashData = (data: any): { isValid: boolean; errors: string[] } => {
    const errors: string[] = [];

    if (data.finalCash < 0) {
      errors.push('Caixa final não pode ser negativo');
    }

    if (data.initialCash < 0) {
      errors.push('Caixa inicial não pode ser negativo');
    }

    if (data.cashSales < 0) {
      errors.push('Vendas em dinheiro não podem ser negativas');
    }

    if (data.cashWithdrawals < 0) {
      errors.push('Sangrias não podem ser negativas');
    }

    return {
      isValid: errors.length === 0,
      errors
    };
  };

  return {
    calculateDivergence,
    classifyDivergenceSeverity,
    generateActionSuggestions,
    validateCashData
  };
};

    API->>DC: calculateShiftDivergence()
    DC->>DB: Buscar vendas em dinheiro
    DC->>DB: Buscar sangrias
    DC->>DC: Aplicar fórmula: Final - (Inicial + Vendas - Sangrias)
    DC->>API: Retornar divergência calculada
    API->>DB: Atualizar turno com divergência
    API->>DB: Inserir timeline de auditoria
    API->>WS: Notificar divergência (se > 0.01)
    WS->>A: Alerta para administradores
    API->>F: Confirmar fechamento
```
