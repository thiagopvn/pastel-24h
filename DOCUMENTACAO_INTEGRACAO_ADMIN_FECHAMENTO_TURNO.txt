# DOCUMENTAÇÃO DETALHADA - INTEGRAÇÃO PAINEL ADMINISTRATIVO E FECHAMENTO DE TURNO

## VISÃO GERAL DA INTEGRAÇÃO

A integração entre o painel administrativo e o fechamento de turno do funcionário representa um dos fluxos mais complexos da plataforma Pastel24h. Esta documentação detalha como as tecnologias se comportam em cada solicitação, desde a interface do funcionário até a validação administrativa, incluindo correções pós-fechamento e auditoria completa.

## ARQUITETURA DE INTEGRAÇÃO

### Fluxo de Dados Bidirecional
```
[Funcionário] ←→ [WebSocket] ←→ [Backend] ←→ [Admin Panel]
     ↓                              ↓            ↓
[Frontend React]              [Express.js]  [Admin React]
     ↓                              ↓            ↓
[TanStack Query] ←→ [API Client] ←→ [SQLite] ←→ [Drizzle ORM]
```

### Tecnologias Envolvidas por Camada

**Frontend (React + TypeScript):**
- React 18 com hooks e context
- TanStack Query para state management
- Wouter para roteamento
- WebSocket para tempo real
- Tailwind CSS + Shadcn/ui

**Backend (Express.js + TypeScript):**
- Express.js com middleware personalizado
- Passport.js para autenticação
- WebSocket Server (ws)
- Drizzle ORM para banco de dados
- Zod para validação de dados

**Banco de Dados (SQLite):**
- SQLite com WAL mode
- Índices otimizados
- Transações ACID
- Triggers para auditoria

```typescript
// server/routes.ts - Endpoint de fechamento
app.post('/api/shifts/:id/close', requireAuth, async (req, res) => {
  const shiftId = parseInt(req.params.id);
  const {
    finalCash,
    finalCoins,
    envelopeCash,
    envelopeCoins,
    payments,
    cumulativeCardValues
  } = req.body;

  // Iniciar transação para garantir consistência
  await db.transaction(async (tx) => {
    try {
      // 1. Verificar autorização e estado do turno
      const shift = await tx.query.shifts.findFirst({
        where: and(eq(shifts.id, shiftId), eq(shifts.status, 'open')),
        with: {
          user: true,
          payments: true,
          records: { with: { product: true } }
        }
      });

      if (!shift) {
        throw new Error('Turno não encontrado ou já fechado');
      }

      // Verificar permissão
      if (req.user.role !== 'admin' && req.user.id !== shift.userId) {
        throw new Error('Sem permissão para fechar este turno');
      }

      // 2. Processar cálculos de cartão se fornecidos
      let finalPayments = payments;

      if (cumulativeCardValues && Object.values(cumulativeCardValues).some(v => v)) {
        const cardCalculator = new CardPaymentCalculator(tx);
        const calculatedCards = await cardCalculator.calculateRealCardPayments(
          shiftId,
          { ...payments, ...cumulativeCardValues }
        );

        finalPayments = {
          cash: payments.cash,
          pix: payments.pix,
          stoneCard: calculatedCards.realValues.stoneCard.toFixed(2),
          stoneVoucher: calculatedCards.realValues.stoneVoucher.toFixed(2),
          pagBankCard: calculatedCards.realValues.pagBankCard.toFixed(2)
        };

        // Atualizar com valores cumulativos
        await tx.update(shiftPayments)
          .set({
            ...finalPayments,
            stoneCardCumulative: cumulativeCardValues.stoneCardCumulative,
            stoneVoucherCumulative: cumulativeCardValues.stoneVoucherCumulative,
            pagBankCardCumulative: cumulativeCardValues.pagBankCardCumulative,
            calculatedFromCumulative: true
          })
          .where(eq(shiftPayments.shiftId, shiftId));
      } else {
        // Atualizar apenas valores reais
        await tx.update(shiftPayments)
          .set(finalPayments)
          .where(eq(shiftPayments.shiftId, shiftId));
      }

      // 3. Calcular valores financeiros
      const actualCash = parseFloat(finalCash) || 0;
      const actualCoins = parseFloat(finalCoins) || 0;
      const envelopeCashAmount = parseFloat(envelopeCash) || 0;
      const envelopeCoinsAmount = parseFloat(envelopeCoins) || 0;

      const cashForNext = actualCash - envelopeCashAmount;
      const coinsForNext = actualCoins - envelopeCoinsAmount;

      // Calcular divergência de caixa
      const initialCash = parseFloat(shift.initialCash || '0');
      const cashSales = parseFloat(finalPayments.cash || '0');

      const cashWithdrawals = await tx.select({ total: sum(cashAdjustments.amount) })
        .from(cashAdjustments)
        .where(and(
          eq(cashAdjustments.shiftId, shiftId),
          eq(cashAdjustments.type, 'withdrawal')
        ));

      const totalWithdrawals = cashWithdrawals[0]?.total || 0;
      const expectedCash = initialCash + cashSales - totalWithdrawals;
      const cashDivergence = actualCash - expectedCash;

      // 4. Fechar turno
      await tx.update(shifts).set({
        status: 'closed',
        endTime: new Date().toISOString(),
        finalCash: actualCash.toFixed(2),
        finalCoins: actualCoins.toFixed(2),
        envelopeCash: envelopeCashAmount.toFixed(2),
        envelopeCoins: envelopeCoinsAmount.toFixed(2),
        cashForNextShift: cashForNext.toFixed(2),
        coinsForNextShift: coinsForNext.toFixed(2)
      }).where(eq(shifts.id, shiftId));

      // 5. Finalizar colaboradores
      await tx.update(shiftCollaborators)
        .set({
          endTime: new Date().toISOString(),
          hoursWorked: sql`(julianday(${new Date().toISOString()}) - julianday(start_time)) * 24`
        })
        .where(and(
          eq(shiftCollaborators.shiftId, shiftId),
          isNull(shiftCollaborators.endTime)
        ));

      // 6. Criar registro de auditoria
      await tx.insert(timeline).values({
        shiftId,
        userId: req.user.id,
        action: 'SHIFT_CLOSED',
        description: `Turno fechado - Divergência: R$${cashDivergence.toFixed(2)}`,
        metadata: JSON.stringify({
          finalCash: actualCash,
          finalCoins: actualCoins,
          cashDivergence,
          payments: finalPayments,
          cumulativeCardValues: cumulativeCardValues || null,
          closedBy: req.user.id,
          closedAt: new Date().toISOString()
        })
      });

      // 7. Notificar administradores via WebSocket
      notifyAdminClients('shift-closed', {
        shiftId,
        userId: shift.userId,
        userName: shift.user.name,
        cashDivergence,
        totalPayments: Object.values(finalPayments).reduce((sum, val) => sum + parseFloat(val), 0),
        closedAt: new Date().toISOString()
      });

      res.json({
        message: 'Turno fechado com sucesso',
        cashDivergence,
        shiftId
      });

    } catch (error) {
      // Rollback automático em caso de erro
      throw error;
    }
  });
});
```

```typescript
// server/ws.ts - Sistema de notificação
export const notifyAdminClients = (event: string, data: any) => {
  if (!global.adminWss) return;

  const notification = {
    type: 'admin-notification',
    event,
    data,
    timestamp: new Date().toISOString()
  };

  global.adminWss.clients.forEach((client: any) => {
    if (client.readyState === WebSocket.OPEN && client.isAdmin) {
      client.send(JSON.stringify(notification));
    }
  });

  // Log para auditoria
  console.log(`Admin notification sent: ${event}`, data);
};

// Configuração de conexão admin
export const setupAdminWebSocket = (wss: WebSocket.Server) => {
  wss.on('connection', (ws: any, req: any) => {
    const url = new URL(req.url!, `http://${req.headers.host}`);
    const isAdmin = url.searchParams.get('role') === 'admin';

    if (isAdmin) {
      ws.isAdmin = true;
      console.log('Admin conectado ao WebSocket');

      // Enviar estado inicial
      ws.send(JSON.stringify({
        type: 'admin-connected',
        timestamp: new Date().toISOString()
      }));
    }
  });
};
```

### 3. BACKEND - SISTEMA DE CORREÇÕES ADMINISTRATIVAS

```typescript
// server/corrections-utils.ts
export class ShiftCorrectionService {
  constructor(private db: any) {}

  async applyCorrection(
    shiftId: number,
    correction: CorrectionData,
    adminUserId: number
  ): Promise<CorrectionResult> {

    return await this.db.transaction(async (tx) => {
      try {
        // 1. Validar turno e permissões
        const shift = await tx.query.shifts.findFirst({
          where: eq(shifts.id, shiftId),
          with: { user: true, payments: true }
        });

        if (!shift) {
          throw new Error('Turno não encontrado');
        }

        if (shift.status === 'open') {
          throw new Error('Não é possível corrigir turno em andamento');
        }

        // 2. Validar dados da correção
        const validationResult = await this.validateCorrection(shift, correction);
        if (!validationResult.isValid) {
          throw new Error(validationResult.error);
        }

        // 3. Backup dos valores atuais
        const currentValue = await this.getCurrentValue(shift, correction.type, correction.field);

        // 4. Aplicar correção baseada no tipo
        let updatedData: any = {};

        switch (correction.type) {
          case 'payment':
            updatedData = await this.applyPaymentCorrection(tx, shiftId, correction);
            break;
          case 'inventory':
            updatedData = await this.applyInventoryCorrection(tx, shiftId, correction);
            break;
          case 'cash':
            updatedData = await this.applyCashCorrection(tx, shiftId, correction);
            break;
        }

        // 5. Registrar correção na tabela de auditoria
        const [correctionRecord] = await tx.insert(corrections).values({
          shiftId,
          adminUserId,
          type: correction.type,
          field: correction.field,
          oldValue: currentValue.toString(),
          newValue: correction.newValue,
          reason: correction.reason,
          appliedAt: new Date().toISOString(),
          metadata: JSON.stringify({
            originalShiftData: updatedData.before,
            correctedShiftData: updatedData.after,
            adminUser: adminUserId,
            timestamp: new Date().toISOString()
          })
        }).returning();

        // 6. Timeline de auditoria
        await tx.insert(timeline).values({
          shiftId,
          userId: adminUserId,
          action: 'ADMIN_CORRECTION',
          description: `Correção aplicada: ${correction.field} ${currentValue} → ${correction.newValue}`,
          metadata: JSON.stringify({
            correctionId: correctionRecord.id,
            type: correction.type,
            field: correction.field,
            reason: correction.reason
          })
        });

        // 7. Recalcular métricas afetadas
        await this.recalculateAffectedMetrics(tx, shiftId, correction.type);

        // 8. Notificar sistemas interessados
        notifyAdminClients('correction-applied', {
          shiftId,
          correctionId: correctionRecord.id,
          type: correction.type,
          field: correction.field,
          adminUserId
        });

        return {
          success: true,
          correctionId: correctionRecord.id,
          oldValue: currentValue,
          newValue: parseFloat(correction.newValue),
          affectedMetrics: updatedData.affectedMetrics
        };

      } catch (error) {
        console.error('Erro ao aplicar correção:', error);
        throw error;
      }
    });
  }

  private async applyPaymentCorrection(
    tx: any,
    shiftId: number,
    correction: CorrectionData
  ): Promise<any> {
    const beforeUpdate = await tx.query.shiftPayments.findFirst({
      where: eq(shiftPayments.shiftId, shiftId)
    });

    // Atualizar pagamento específico
    await tx.update(shiftPayments)
      .set({ [correction.field]: correction.newValue })
      .where(eq(shiftPayments.shiftId, shiftId));

    const afterUpdate = await tx.query.shiftPayments.findFirst({
      where: eq(shiftPayments.shiftId, shiftId)
    });

    // Recalcular divergência de caixa se necessário
    if (correction.field === 'cash') {
      await this.recalculateCashDivergence(tx, shiftId);
    }

    return {
      before: beforeUpdate,
      after: afterUpdate,
      affectedMetrics: ['totalPayments', 'cashDivergence']
    };
  }

  private async recalculateCashDivergence(tx: any, shiftId: number): Promise<void> {
    // Buscar dados atuais do turno
    const shiftData = await tx.query.shifts.findFirst({
      where: eq(shifts.id, shiftId),
      with: { payments: true }
    });

    if (!shiftData) return;

    // Buscar sangrias
    const withdrawalsResult = await tx.select({ total: sum(cashAdjustments.amount) })
      .from(cashAdjustments)
      .where(and(
        eq(cashAdjustments.shiftId, shiftId),
        eq(cashAdjustments.type, 'withdrawal')
      ));

    const totalWithdrawals = withdrawalsResult[0]?.total || 0;

    // Calcular nova divergência
    const initialCash = parseFloat(shiftData.initialCash || '0');
    const finalCash = parseFloat(shiftData.finalCash || '0');
    const cashSales = parseFloat(shiftData.payments?.cash || '0');

    const expectedCash = initialCash + cashSales - totalWithdrawals;
    const newDivergence = finalCash - expectedCash;

    // Atualizar divergência calculada
    await tx.update(shifts)
      .set({
        calculatedCashDivergence: newDivergence.toFixed(2),
        lastCorrectionAt: new Date().toISOString()
      })
      .where(eq(shifts.id, shiftId));
  }
}
```

```typescript
// server/ws-advanced.ts
export class AdvancedWebSocketServer {
  private wss: WebSocket.Server;
  private channels: Map<string, Set<WebSocket>> = new Map();
  private clientMetadata: WeakMap<WebSocket, ClientMetadata> = new WeakMap();

  constructor(server: any) {
    this.wss = new WebSocket.Server({
      server,
      path: '/ws-api'
    });

    this.setupConnectionHandling();
  }

  private setupConnectionHandling() {
    this.wss.on('connection', (ws: WebSocket, req: any) => {
      const url = new URL(req.url!, `http://${req.headers.host}`);
      const role = url.searchParams.get('role');
      const shiftId = url.searchParams.get('shiftId');
      const userId = url.searchParams.get('userId');

      // Metadados do cliente
      const metadata: ClientMetadata = {
        role: role as 'admin' | 'employee',
        shiftId: shiftId ? parseInt(shiftId) : null,
        userId: userId ? parseInt(userId) : null,
        connectedAt: new Date(),
        lastActivity: new Date()
      };

      this.clientMetadata.set(ws, metadata);

      // Inscrever em canais apropriados
      this.subscribeToChannels(ws, metadata);

      // Handlers de mensagem
      ws.on('message', (data: string) => {
        this.handleMessage(ws, data);
      });

      ws.on('close', () => {
        this.unsubscribeFromAllChannels(ws);
      });

      ws.on('pong', () => {
        const meta = this.clientMetadata.get(ws);
        if (meta) {
          meta.lastActivity = new Date();
        }
      });

      // Enviar confirmação de conexão
      this.sendToClient(ws, {
        type: 'connection-established',
        channels: this.getClientChannels(ws),
        metadata
      });
    });

    // Heartbeat para manter conexões vivas
    setInterval(() => {
      this.wss.clients.forEach((ws: WebSocket) => {
        if (ws.readyState === WebSocket.OPEN) {
          ws.ping();
        }
      });
    }, 30000);
  }

  private subscribeToChannels(ws: WebSocket, metadata: ClientMetadata) {
    // Canal global
    this.subscribeToChannel(ws, 'global');

    // Canal baseado em role
    if (metadata.role) {
      this.subscribeToChannel(ws, `role:${metadata.role}`);
    }

    // Canal de turno específico
    if (metadata.shiftId) {
      this.subscribeToChannel(ws, `shift:${metadata.shiftId}`);
    }

    // Canal de usuário específico
    if (metadata.userId) {
      this.subscribeToChannel(ws, `user:${metadata.userId}`);
    }

    // Canal administrativo (apenas para admins)
    if (metadata.role === 'admin') {
      this.subscribeToChannel(ws, 'admin-notifications');
      this.subscribeToChannel(ws, 'admin-alerts');
    }
  }

  private subscribeToChannel(ws: WebSocket, channel: string) {
    if (!this.channels.has(channel)) {
      this.channels.set(channel, new Set());
    }
    this.channels.get(channel)!.add(ws);
  }

  private unsubscribeFromAllChannels(ws: WebSocket) {
    this.channels.forEach((clients, channel) => {
      clients.delete(ws);
      if (clients.size === 0) {
        this.channels.delete(channel);
      }
    });
    this.clientMetadata.delete(ws);
  }

  // Método público para enviar para canais específicos
  public broadcastToChannel(channel: string, message: any) {
    const clients = this.channels.get(channel);
    if (!clients) return;

    const payload = JSON.stringify({
      ...message,
      channel,
      timestamp: new Date().toISOString()
    });

    clients.forEach((client: WebSocket) => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(payload);
      }
    });
  }

  // Notificação específica para fechamento de turno
  public notifyShiftClosure(shiftData: any) {
    // Para funcionários do turno
    this.broadcastToChannel(`shift:${shiftData.shiftId}`, {
      type: 'shift-closed',
      data: shiftData
    });

    // Para administradores
    this.broadcastToChannel('admin-notifications', {
      type: 'shift-closure-notification',
      priority: shiftData.cashDivergence !== 0 ? 'high' : 'normal',
      data: shiftData
    });

    // Para canal global (estatísticas gerais)
    this.broadcastToChannel('global', {
      type: 'shift-statistics-updated',
      data: {
        totalShiftsClosed: shiftData.totalShiftsClosed,
        dailyRevenue: shiftData.dailyRevenue
      }
    });
  }

  private sendToClient(ws: WebSocket, message: any) {
    if (ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify({
        ...message,
        timestamp: new Date().toISOString()
      }));
    }
  }
}
```

```typescript
// client/src/hooks/useAdvancedWebSocket.ts
export const useAdvancedWebSocket = (options: WebSocketOptions) => {
  const [connectionStatus, setConnectionStatus] = useState<'connecting' | 'connected' | 'disconnected'>('disconnected');
  const [lastMessage, setLastMessage] = useState<any>(null);
  const [messageHistory, setMessageHistory] = useState<any[]>([]);

  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimeoutRef = useRef<NodeJS.Timeout | null>(null);
  const heartbeatIntervalRef = useRef<NodeJS.Interval | null>(null);

  const queryClient = useQueryClient();

  const connect = useCallback(() => {
    if (wsRef.current?.readyState === WebSocket.OPEN) return;

    setConnectionStatus('connecting');

    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const params = new URLSearchParams({
      role: options.role,
      ...(options.shiftId && { shiftId: options.shiftId.toString() }),
      ...(options.userId && { userId: options.userId.toString() })
    });

    const wsUrl = `${protocol}//${window.location.host}/ws-api?${params}`;

    wsRef.current = new WebSocket(wsUrl);

    wsRef.current.onopen = () => {
      console.log('WebSocket conectado');
      setConnectionStatus('connected');

      // Iniciar heartbeat
      heartbeatIntervalRef.current = setInterval(() => {
        if (wsRef.current?.readyState === WebSocket.OPEN) {
          wsRef.current.send(JSON.stringify({ type: 'ping' }));
        }
      }, 30000);
    };

    wsRef.current.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data);
        setLastMessage(message);
        setMessageHistory(prev => [message, ...prev.slice(0, 99)]); // Manter últimas 100

        // Processar mensagem baseada no tipo
        handleMessage(message);
      } catch (error) {
        console.error('Erro ao processar mensagem WebSocket:', error);
      }
    };

    wsRef.current.onclose = (event) => {
      console.log('WebSocket desconectado:', event.code, event.reason);
      setConnectionStatus('disconnected');

      // Limpar heartbeat
      if (heartbeatIntervalRef.current) {
        clearInterval(heartbeatIntervalRef.current);
      }

      // Tentar reconectar se não foi fechamento intencional
      if (event.code !== 1000) {
        scheduleReconnect();
      }
    };

    wsRef.current.onerror = (error) => {
      console.error('Erro WebSocket:', error);
      setConnectionStatus('disconnected');
    };
  }, [options]);

  const handleMessage = (message: any) => {
    switch (message.type) {
      case 'shift-closed':
        handleShiftClosed(message.data);
        break;

      case 'admin-notification':
        handleAdminNotification(message);
        break;

      case 'correction-applied':
        handleCorrectionApplied(message.data);
        break;

      case 'inventory-updated':
        handleInventoryUpdated(message.data);
        break;
    }
  };

  const handleShiftClosed = (shiftData: any) => {
    // Invalidar queries relacionadas
    queryClient.invalidateQueries(['shift', shiftData.shiftId]);
    queryClient.invalidateQueries(['shifts']);

    if (options.role === 'admin') {
      queryClient.invalidateQueries(['admin-dashboard']);
      queryClient.invalidateQueries(['daily-stats']);
    }

    // Notificação visual
    if (shiftData.cashDivergence !== 0) {
      toast.warning(`Divergência detectada: R$ ${shiftData.cashDivergence.toFixed(2)}`);
    }

    // Callback personalizado
    options.onShiftClosed?.(shiftData);
  };

  const handleCorrectionApplied = (correctionData: any) => {
    queryClient.invalidateQueries(['admin-shift-details', correctionData.shiftId]);
    queryClient.invalidateQueries(['shift-corrections', correctionData.shiftId]);

    toast.success('Correção aplicada com sucesso');
    options.onCorrectionApplied?.(correctionData);
  };

  const scheduleReconnect = () => {
    if (reconnectTimeoutRef.current) {
      clearTimeout(reconnectTimeoutRef.current);
    }

    reconnectTimeoutRef.current = setTimeout(() => {
      connect();
    }, 3000); // Tentar reconectar em 3 segundos
  };

  const sendMessage = (message: any) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({
        ...message,
        timestamp: new Date().toISOString()
      }));
      return true;
    }
    return false;
  };

  const disconnect = () => {
    if (reconnectTimeoutRef.current) {
      clearTimeout(reconnectTimeoutRef.current);
    }

    if (heartbeatIntervalRef.current) {
      clearInterval(heartbeatIntervalRef.current);
    }

    if (wsRef.current) {
      wsRef.current.close(1000, 'Desconexão intencional');
    }
  };

  // Conectar ao montar, desconectar ao desmontar
  useEffect(() => {
    connect();
    return disconnect;
  }, [connect]);

  return {
    connectionStatus,
    lastMessage,
    messageHistory,
    sendMessage,
    reconnect: connect,
    disconnect
  };
};
```

```typescript
// server/audit-system.ts
export class AuditSystem {
  constructor(private db: any) {}

  async logShiftClosure(shiftId: number, closureData: any, userId: number) {
    const auditEntry = {
      event: 'SHIFT_CLOSURE',
      shiftId,
      userId,
      timestamp: new Date().toISOString(),
      data: {
        financialData: closureData.payments,
        cumulativeValues: closureData.cumulativeCardValues,
        cashData: {

          final: closureData.finalCash,
          divergence: closureData.cashDivergence
        },

        systemCalculations: closureData.systemCalculations
      },
      checksum: this.generateChecksum(closureData)
    };

    await this.db.insert(auditLog).values(auditEntry);

    // Verificar se há padrões suspeitos
    await this.detectAnomalies(shiftId, closureData);
  }

  async logAdminCorrection(correctionData: any, adminUserId: number) {
    const auditEntry = {
      event: 'ADMIN_CORRECTION',
      shiftId: correctionData.shiftId,
      userId: adminUserId,
      timestamp: new Date().toISOString(),
      data: {
        correction: correctionData,
        beforeState: correctionData.beforeState,
        afterState: correctionData.afterState,
        reason: correctionData.reason
      },
      checksum: this.generateChecksum(correctionData)
    };

    await this.db.insert(auditLog).values(auditEntry);

    // Notificar sobre correção administrativa
    await this.notifyComplianceTeam('ADMIN_CORRECTION', auditEntry);
  }

  private async detectAnomalies(shiftId: number, data: any) {
    const anomalies: string[] = [];

    // Verificar divergência de caixa alta
    if (Math.abs(data.cashDivergence) > 50) {
      anomalies.push(`HIGH_CASH_DIVERGENCE: R$ ${data.cashDivergence}`);
    }

    // Verificar padrão de fechamentos
    const recentShifts = await this.getRecentShifts(data.userId, 5);
    const divergences = recentShifts.map(s => Math.abs(s.cashDivergence || 0));
    const avgDivergence = divergences.reduce((a, b) => a + b, 0) / divergences.length;

    if (avgDivergence > 20) {
      anomalies.push(`PATTERN_HIGH_DIVERGENCE: Média ${avgDivergence.toFixed(2)}`);
    }

    // Verificar horários incomuns
    const closureTime = new Date(data.closureTime);
    const hour = closureTime.getHours();

    if (hour < 6 || hour > 23) {
      anomalies.push(`UNUSUAL_CLOSURE_TIME: ${hour}:${closureTime.getMinutes()}`);
    }

    if (anomalies.length > 0) {
      await this.flagForReview(shiftId, anomalies);
    }
  }

  private generateChecksum(data: any): string {
    return crypto
      .createHash('sha256')
      .update(JSON.stringify(data))
      .digest('hex');
  }
```
